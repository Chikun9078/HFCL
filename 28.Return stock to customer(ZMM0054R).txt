FORM get_data.
  IF r_cust = 'X'.
    SELECT    a~zstk_crt_id
              a~item_no
              a~werks
              a~zstrloc
              a~matnr
              a~pspel
              a~zbusptr
              a~zqty
              a~uom
              a~zmode
              a~remarks
              a~sydate
              b~maktx
        FROM zmm_stk_entry AS a
        LEFT OUTER JOIN makt AS b ON b~matnr EQ a~matnr AND
                                     b~spras EQ sy-langu
        INTO TABLE lt_final
        WHERE zmode = 'MANUAL'
        AND a~sydate  IN s_date
        AND a~werks   IN s_werks
        AND a~zstrloc IN s_lgort
        AND a~matnr   IN s_matnr.

  ELSEIF r_vendor = 'X'.
    SELECT    a~zstk_trf_id
              a~zstk_trf_line
              a~zstk_crt_id
              a~item_no
              a~werks
              a~zstrloc
              a~customer
              a~vendor
              a~matnr
              a~pspel
              a~qty
              a~uom
              a~site_name
              b~sydate
              c~maktx
              d~name1 AS Vendor_name
       INTO TABLE lt_final_v
       FROM zmm_stk_transfer AS a
       INNER JOIN  zmm_stk_entry AS b ON  a~zstk_crt_id = b~zstk_crt_id AND a~item_no =  b~item_no
       LEFT OUTER JOIN makt AS c ON c~matnr EQ a~matnr AND c~spras EQ sy-langu
       LEFT OUTER JOIN lfa1 AS d ON d~lifnr EQ a~vendor
       WHERE b~zmode = 'MANUAL'
       AND b~sydate  IN s_date
       AND b~werks   IN s_werks
       AND b~zstrloc IN s_lgort
       AND b~matnr   IN s_matnr
       AND a~vendor  IN s_vendor.
  ENDIF.
ENDFORM.

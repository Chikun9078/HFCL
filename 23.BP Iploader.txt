FORM bdc_dynpro USING program dynpro.
  CLEAR bdcdata.
  bdcdata-program  = program.
  bdcdata-dynpro   = dynpro.
  bdcdata-dynbegin = 'X'.
  APPEND bdcdata.

*----------------------------------------------------------------------*
*        Insert field                                                  *
*----------------------------------------------------------------------*
FORM bdc_field USING fnam fval.
  IF fval <> space.
    CLEAR bdcdata.
    bdcdata-fnam = fnam.
    bdcdata-fval = fval.
    CONDENSE bdcdata-fval.
    APPEND bdcdata.
  ENDIF.
ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  SUCCESS_FILE
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM success_file .

  LOOP AT messtab WHERE msgtyp = 'S' AND msgnr = '312'.
    CLEAR: l_msg1.
    CALL FUNCTION 'FORMAT_MESSAGE'
      EXPORTING
        id        = messtab-msgid
        lang      = sy-langu
        no        = messtab-msgnr
        v1        = messtab-msgv1
        v2        = messtab-msgv2
        v3        = messtab-msgv3
        v4        = messtab-msgv4
      IMPORTING
        msg       = l_msg1
      EXCEPTIONS ##FM_SUBRC_OK
        not_found = 1
        OTHERS    = 2.
    ADD 1 TO l_suc.
    MOVE l_msg1 TO l_sucmsg.

    t_success-sr  = l_suc.
    t_success-msg = l_sucmsg.
    APPEND t_success.
  ENDLOOP.

ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  ERROR_FILE
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM error_file .
  LOOP AT messtab WHERE msgtyp = 'E'.
    CLEAR: l_msg.
    CALL FUNCTION 'FORMAT_MESSAGE'
      EXPORTING
        id        = messtab-msgid
        lang      = sy-langu
        no        = messtab-msgnr
        v1        = messtab-msgv1
        v2        = messtab-msgv2
        v3        = messtab-msgv3
        v4        = messtab-msgv4
      IMPORTING
        msg       = l_msg
      EXCEPTIONS ##FM_SUBRC_OK
        not_found = 1
        OTHERS    = 2.
    ADD 1 TO l_er.
    MOVE l_msg TO l_ermsg.

    t_error-sr = l_er.
    t_error-msg = l_ermsg.
    APPEND t_error.
  ENDLOOP.
ENDFORM.

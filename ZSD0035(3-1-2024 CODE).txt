FORM get_raw_sale_ordr_data .

  CLEAR : raw_sale_order_data[].

  SELECT        vbak~vbeln
                vbap~posnr
                vbep~etenr
                vbak~audat
                vbak~auart
                vbak~bstnk
                vbak~kunnr
                vbak~bstdk
                vbak~vkgrp
                vbak~vkorg
                vbak~vkbur
                vbap~matnr
                vbap~arktx
                vbap~vrkme
                vbap~kwmeng
                vbap~uebto
                vbap~untto
                vbap~werks
                vbap~ntgew
                vbap~gewei
                vbap~netpr
                vbap~gbsta
                vbap~lfsta
                vbep~edatu
                vbak~waerk
                vbap~zcrd
                vbap~zspd
                vbap~ztds
                vbap~tdsvr
                vbap~zdrm
                vbap~zpds
                vbap~zduppr
                vbap~zdlow
                vbap~zshl
                vbap~zremk
                vbap~ztol
                vbap~zdtol
                vbap~zotol
                vbap~knumv_ana
    INTO        TABLE raw_sale_order_data
    FROM        vbak  INNER JOIN vbap
                      ON vbak~vbeln = vbap~vbeln
                      LEFT JOIN vbep
                      ON vbap~vbeln = vbep~vbeln AND
                         vbap~posnr = vbep~posnr
    WHERE       vbak~vbeln    IN  s_vbeln  AND
                vbak~audat    IN  s_audat  AND
                vbap~werks    IN  s_werks  AND
                vbak~auart    IN  s_auart  AND
                vbak~vkorg    IN  s_vkorg  AND
                vbak~kunnr    IN  s_kunnr  AND
                vbak~vkbur    IN  s_vkbur  AND
                vbak~vkgrp    IN  s_vkgrp  AND
                vbak~bstnk    IN  s_bstnk.

  SORT raw_sale_order_data BY vbeln posnr edatu DESCENDING.
  DELETE ADJACENT DUPLICATES FROM raw_sale_order_data COMPARING vbeln posnr.
ENDFORM.

             vbkd~posnr
             vbkd~zterm
             vbkd~inco1

    INTO     TABLE raw_sordr_business_data

*& <--  p2        text

FORM conversion USING uv_qty TYPE any
                      uv_uom TYPE any
                CHANGING cv_qty TYPE any.
  CALL FUNCTION 'UNIT_CONVERSION_SIMPLE'
    EXPORTING
      input                = uv_qty
      no_type_check        = 'X'
      round_sign           = ' '
      unit_in              = uv_uom
      unit_out             = 'KM'
    IMPORTING
      output               = cv_qty
    EXCEPTIONS
      conversion_not_found = 1
      division_by_zero     = 2
      input_invalid        = 3
      output_invalid       = 4
      overflow             = 5
      type_invalid         = 6
      units_missing        = 7
      unit_in_not_found    = 8
      unit_out_not_found   = 9
      OTHERS               = 10.
ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_raw_Pricing_Elements
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_raw_Pricing_Elements .
  IF raw_sale_order_data[] IS NOT INITIAL.
    SELECT knumv, kposn, kschl, kbetr
    FROM prcd_elements INTO TABLE @raw_Pricing_Elements[]
    FOR ALL ENTRIES IN @raw_sale_order_data
    WHERE knumv = @raw_sale_order_data-knumv_ana AND
          kposn = @raw_sale_order_data-posnr     AND
          kschl IN ('ZBAS','ZBPS').
  ENDIF.

  SORT  raw_Pricing_Elements[] BY knumv kposn  ASCENDING.
ENDFORM.

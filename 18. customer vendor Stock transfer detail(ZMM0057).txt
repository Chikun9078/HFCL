FORM get_stk_entry_transfer_data .
  SELECT    zmm_stk_entry~zstk_crt_id,
            zmm_stk_entry~item_no,
            zmm_stk_transfer~zstk_trf_id ,
            zmm_stk_transfer~zstk_trf_line,
            zmm_stk_entry~werks,
            zmm_stk_entry~zstrloc,
            zmm_stk_entry~matnr,
            makt~maktx,
            zmm_stk_entry~zbusptr,
            zmm_stk_entry~pspel,
            zmm_stk_entry~zqty,
            zmm_stk_entry~uom,
            zmm_stk_entry~del_no,
            zmm_stk_entry~bill_no,
            zmm_stk_entry~zmode,
            zmm_stk_entry~remarks,
            zmm_stk_entry~upload_date,
            zmm_stk_transfer~stock_issue_dt,
            zmm_stk_transfer~document_date,
            zmm_stk_transfer~vendor,
            zmm_stk_transfer~qty,
            zmm_stk_transfer~document_no,
            zmm_stk_transfer~site_name
    FROM zmm_stk_entry
     LEFT JOIN  makt ON  zmm_stk_entry~matnr = makt~matnr AND
                         makt~spras          = @sy-langu
    LEFT JOIN  zmm_stk_transfer ON  zmm_stk_entry~zstk_crt_id = zmm_stk_transfer~zstk_crt_id AND
                                    zmm_stk_entry~item_no     = zmm_stk_transfer~item_no
    INTO TABLE @lt_entry_trans                                                                              ##TOO_MANY_ITAB_FIELDS
    WHERE zmm_stk_entry~werks     IN @s_plant  AND
          zmm_stk_entry~matnr     IN @s_matnr  AND
          zmm_stk_entry~zbusptr   IN @s_cust   AND
          zmm_stk_entry~pspel     IN @s_wbs    AND
          zmm_stk_transfer~vendor IN @s_vendor.

  IF sy-subrc EQ 0.
    LOOP AT lt_entry_trans  ASSIGNING FIELD-SYMBOL(<fs_final>).
      AUTHORITY-CHECK OBJECT 'ZMMA_CDEV'
      ID 'WERKS' FIELD <fs_final>-werks
      ID 'TCODE' FIELD 'ZMM0057'.
      IF sy-subrc <> 0.
        CLEAR:<fs_final>-werks.
      ENDIF.
    ENDLOOP.
    DELETE lt_entry_trans  WHERE werks IS INITIAL.
    IF lt_entry_trans IS INITIAL.
      MESSAGE e166(zmm).
    ENDIF.
  ENDIF.

  SORT lt_entry_trans BY zstk_crt_id item_no.
ENDFORM.

FORM get_stk_settle_data .
  IF lt_entry_trans IS NOT INITIAL.
    SELECT zstk_stl_id,zstk_stl_line,
           zstk_trf_id,zstk_trf_line,
           matnr,
           consum_qty,faulty_qty ,
           return_qty,missing_qty,
           sydate,sytime
    FROM zmm_stk_settle
    INTO TABLE @lt_stk_settle
    FOR ALL ENTRIES IN @lt_entry_trans
    WHERE zstk_trf_id   = @lt_entry_trans-zstk_trf_id AND
          zstk_trf_line = @lt_entry_trans-zstk_trf_line.
  ENDIF.

  SORT lt_stk_settle BY zstk_trf_id zstk_trf_line matnr sydate sytime .
ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_stk_settle_data_summerized
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_stk_settle_data_summerized .
  IF lt_entry_trans IS NOT INITIAL.
    SELECT zstk_trf_id,
           zstk_trf_line,
           matnr,
          SUM( consum_qty )  AS consum_qty ,
          SUM( faulty_qty )  AS faulty_qty  ,
          SUM( return_qty )  AS return_qty,
          SUM( missing_qty ) AS missing_qty
    FROM zmm_stk_settle
    INTO CORRESPONDING FIELDS OF TABLE @lt_stk_settle_smrz
    WHERE vendor in @s_vendor and
          pspel  in @s_wbs    and
          matnr  in @s_matnr
    GROUP BY zstk_trf_id, zstk_trf_line, matnr  .
  ENDIF.

  SORT lt_stk_settle_smrz BY zstk_trf_id zstk_trf_line matnr .
ENDFORM.

FORM  process_detail_data.
  CLEAR: ls_entry_trans,ls_stk_settle,lv_balance_qty.

  LOOP AT lt_entry_trans INTO ls_entry_trans.
    ls_final-zstk_crt_id   = ls_entry_trans-zstk_crt_id.
    ls_final-item_no       = ls_entry_trans-item_no .
    ls_final-zstk_trf_id   = ls_entry_trans-zstk_trf_id .
    ls_final-zstk_trf_line = ls_entry_trans-zstk_trf_line .
    ls_final-werks         = ls_entry_trans-werks .
    ls_final-zstrloc       = ls_entry_trans-zstrloc .
    ls_final-matnr         = ls_entry_trans-matnr .
    ls_final-mat_desc      = ls_entry_trans-mat_desc .
    ls_final-zbusptr       = ls_entry_trans-zbusptr .

    SELECT SINGLE kunnr,name1 FROM kna1 INTO @DATA(ls_kna1) WHERE kunnr EQ @ls_entry_trans-zbusptr.
    ls_final-cust_name = ls_kna1-name1 .
    ls_final-pspel     = ls_entry_trans-pspel .
    ls_final-zqty      = ls_entry_trans-zqty .
    ls_final-uom       = ls_entry_trans-uom .
    ls_final-del_no    = ls_entry_trans-del_no .
    ls_final-bill_no   = ls_entry_trans-bill_no .
    ls_final-zmode     = ls_entry_trans-zmode .
    ls_final-remarks   = ls_entry_trans-remarks .

    ls_final-upload_date    = ls_entry_trans-upload_date .
    ls_final-stock_issue_dt = ls_entry_trans-stock_issue_dt .
    ls_final-document_date  = ls_entry_trans-document_date .
    ls_final-vendor         = ls_entry_trans-vendor .

    SELECT SINGLE lifnr,name1 FROM lfa1 INTO @DATA(ls_lfa1) WHERE lifnr EQ @ls_entry_trans-vendor.
    ls_final-vendor_name = ls_lfa1-name1 .
    ls_final-document_no = ls_entry_trans-document_no .
    ls_final-site_name   = ls_entry_trans-site_name .
*    ls_final-qty         = ls_entry_trans-qty .

    IF  line_exists( lt_stk_settle[ zstk_trf_id   = ls_entry_trans-zstk_trf_id  zstk_trf_line = ls_entry_trans-zstk_trf_line matnr = ls_entry_trans-matnr ] ).
      LOOP AT lt_stk_settle INTO ls_stk_settle WHERE zstk_trf_id   = ls_entry_trans-zstk_trf_id   AND
                                                     zstk_trf_line = ls_entry_trans-zstk_trf_line AND
                                                     matnr         = ls_entry_trans-matnr.
*        AT NEW zstk_trf_id.
*          ls_final-qty         = ls_entry_trans-qty .
*        ENDAT.

        ls_final-consum_qty        = ls_stk_settle-consum_qty.
        ls_final-faulty_qty        = ls_stk_settle-faulty_qty.
        ls_final-missing_qty       = ls_stk_settle-missing_qty.
        ls_final-return_qty        = ls_stk_settle-return_qty.
        ls_final-sydate            = ls_stk_settle-sydate.
        ls_final-sytime            = ls_stk_settle-sytime.

        "--> calculating balance quantity..
        IF lv_balance_qty IS INITIAL.
          lv_balance_qty = ls_entry_trans-qty - ( ls_stk_settle-consum_qty + ls_stk_settle-faulty_qty + ls_stk_settle-missing_qty + ls_stk_settle-return_qty ).
        ELSE.
          lv_balance_qty = lv_balance_qty - ( ls_stk_settle-consum_qty + ls_stk_settle-faulty_qty + ls_stk_settle-missing_qty + ls_stk_settle-return_qty ).
        ENDIF.
        ls_final-balance_qty       = lv_balance_qty.

*        AT END OF zstk_trf_id.
*          ls_final-balance_qty_final = lv_balance_qty.
*        ENDAT.

        APPEND ls_final TO lt_final.
        CLEAR:ls_final-consum_qty ,ls_final-faulty_qty,ls_final-missing_qty ,
              ls_final-return_qty,ls_final-balance_qty,ls_final-balance_qty_final,
               ls_final-qty,ls_final-sydate,ls_final-sytime,ls_stk_settle.
      ENDLOOP.
      CLEAR lv_balance_qty.

    ELSE.
      APPEND ls_final TO lt_final..
    ENDIF.
    CLEAR: ls_final,ls_entry_trans.
  ENDLOOP.

  SORT lt_final BY zstk_trf_id zstk_trf_line matnr sydate sytime.

  CLEAR ls_final.
  LOOP AT lt_final ASSIGNING FIELD-SYMBOL(<fs_final>).
    AT NEW zstk_trf_id.
      READ TABLE lt_entry_trans INTO DATA(ls_temp) WITH KEY  zstk_trf_id   = <fs_final>-zstk_trf_id
                                                             zstk_trf_line = <fs_final>-zstk_trf_line
                                                             matnr         = <fs_final>-matnr.
      IF sy-subrc EQ 0.
        <fs_final>-qty = ls_temp-qty.
      ENDIF.
    ENDAT.
    AT END OF zstk_trf_id.
      <fs_final>-balance_qty_final = <fs_final>-balance_qty.
    ENDAT.
  ENDLOOP.
ENDFORM.

    ls_summerized-werks         = ls_entry_trans-werks .
    ls_summerized-zstrloc       = ls_entry_trans-zstrloc .

    ls_summerized-mat_desc      = ls_entry_trans-mat_desc .
    ls_summerized-zbusptr       = ls_entry_trans-zbusptr .

    SELECT SINGLE kunnr,name1 FROM kna1 INTO @DATA(ls_kna1) WHERE kunnr EQ @ls_entry_trans-zbusptr.
    ls_summerized-cust_name = ls_kna1-name1 .
    ls_summerized-pspel     = ls_entry_trans-pspel .
    ls_summerized-zqty      = ls_entry_trans-zqty .
    ls_summerized-uom       = ls_entry_trans-uom .
    ls_summerized-del_no    = ls_entry_trans-del_no .
    ls_summerized-bill_no   = ls_entry_trans-bill_no .
    ls_summerized-zmode     = ls_entry_trans-zmode .
    ls_summerized-remarks   = ls_entry_trans-remarks .

    ls_summerized-upload_date    = ls_entry_trans-upload_date .
    ls_summerized-stock_issue_dt = ls_entry_trans-stock_issue_dt .
    ls_summerized-document_date  = ls_entry_trans-document_date .
    ls_summerized-vendor         = ls_entry_trans-vendor .

    SELECT SINGLE lifnr,name1 FROM lfa1 INTO @DATA(ls_lfa1) WHERE lifnr EQ @ls_entry_trans-vendor.
    ls_summerized-vendor_name = ls_lfa1-name1 .
    ls_summerized-document_no = ls_entry_trans-document_no .
    ls_summerized-site_name   = ls_entry_trans-site_name .
    ls_summerized-qty         = ls_entry_trans-qty .

    READ TABLE lt_stk_settle_smrz INTO DATA(ls_stk_settle_smrz) WITH  KEY zstk_trf_id   = ls_entry_trans-zstk_trf_id
                                                                          zstk_trf_line = ls_entry_trans-zstk_trf_line
                                                                          matnr         = ls_entry_trans-matnr.
    IF sy-subrc EQ 0.
      ls_summerized-consum_qty        = ls_stk_settle_smrz-consum_qty.
      ls_summerized-faulty_qty        = ls_stk_settle_smrz-faulty_qty.
      ls_summerized-missing_qty       = ls_stk_settle_smrz-missing_qty.
      ls_summerized-return_qty        = ls_stk_settle_smrz-return_qty.
    ENDIF.
    lv_balance_temp = ( ls_stk_settle_smrz-consum_qty + ls_stk_settle_smrz-faulty_qty + ls_stk_settle_smrz-missing_qty + ls_stk_settle_smrz-return_qty ).
    ls_summerized-balance_qty = ls_entry_trans-qty - lv_balance_temp .

    APPEND ls_summerized TO lt_final.
    CLEAR : ls_summerized,ls_entry_trans,ls_stk_settle_smrz.
  ENDLOOP.

  SORT lt_final BY zstk_trf_id.
ENDFORM.

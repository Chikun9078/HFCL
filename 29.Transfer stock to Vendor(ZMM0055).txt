FORM get_data .
  SELECT zstk_crt_id item_no werks zstrloc matnr zbusptr pspel zqty uom
   FROM zmm_stk_entry
   INTO TABLE lt_final                    ##TOO_MANY_ITAB_FIELDS
      WHERE werks   IN p_plant AND
            zstrloc IN p_strloc AND
            zbusptr IN p_cust AND
            pspel   IN p_wbs AND
            matnr   IN p_matnr  .

  IF sy-subrc = 0.

    LOOP AT lt_final ASSIGNING FIELD-SYMBOL(<fs_final>).
      AUTHORITY-CHECK OBJECT 'ZMMA_CDEV'
       ID 'WERKS' FIELD <fs_final>-werks
       ID 'TCODE' FIELD 'ZMM0055'.
      IF sy-subrc <> 0.
        CLEAR:<fs_final>-werks.
      ENDIF.

    ENDLOOP.

    DELETE lt_final WHERE werks IS INITIAL.

    IF lt_final IS INITIAL.
      MESSAGE e166(zmm).
    ENDIF.

    LOOP AT lt_final INTO ls_final.

      "-->Read material desc. .
      READ TABLE lt_makt INTO DATA(ls_makt) WITH KEY matnr = ls_final-matnr.
      IF sy-subrc = 0.
        ls_final-mat_desc = ls_makt-maktx.
        MODIFY lt_final FROM ls_final TRANSPORTING mat_desc.
      ENDIF.
      "-->Read customer name.
      READ TABLE lt_kna1 INTO DATA(ls_kna1) WITH KEY kunnr = |{ ls_final-zbusptr ALPHA = IN }|.
      IF sy-subrc = 0.
        ls_final-zbusptr_name = ls_kna1-name1.
        MODIFY lt_final FROM ls_final TRANSPORTING zbusptr_name.
      ENDIF.

      SELECT SINGLE zqty FROM zmm_stk_entry INTO @DATA(ls_stk_entry) WHERE zstk_crt_id = @ls_final-zstk_crt_id AND item_no = @ls_final-item_no. "#EC CI_ALL_FIELDS_NEEDED.
      SELECT SUM( qty ) FROM zmm_stk_transfer INTO @DATA(lv_sumqty) WHERE zstk_crt_id = @ls_final-zstk_crt_id AND item_no = @ls_final-item_no.
      IF sy-subrc = 0.
        ls_final-qty_balance = ls_stk_entry - lv_sumqty.
        CLEAR lv_sumqty.
        MODIFY lt_final FROM ls_final TRANSPORTING qty_balance.
      ENDIF.
    ENDLOOP.

    LOOP AT lt_final INTO ls_final WHERE qty_balance = 0 ##INTO_OK.
      DELETE lt_final INDEX sy-tabix.
    ENDLOOP.

  ENDIF.

ENDFORM.

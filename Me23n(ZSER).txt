FORM entry_neu USING ent_retco TYPE any ent_screen TYPE any ##CALLED.

  TABLES: nast.

  DATA: l_druvo       TYPE druvo,
        l_nast        TYPE nast ##NEEDED,
        l_from_memory  ##TYPE_CHAIN_OK,
        l_doc         TYPE meein_purchase_doc_print ##NEEDED.

  CLEAR ent_retco.

  IF nast-aende EQ space.
    l_druvo = '1'.
  ELSE.
    l_druvo = '2'.
  ENDIF.

  CALL FUNCTION 'ME_READ_PO_FOR_PRINTING'
    EXPORTING
      ix_nast        = nast
      ix_screen      = ent_screen
    IMPORTING
      ex_retco       = ent_retco
      ex_nast        = l_nast
      doc            = l_doc
    CHANGING
      cx_druvo       = l_druvo
      cx_from_memory = l_from_memory.

  CHECK ent_retco EQ 0.
  p_ebeln = nast-objky.

  PERFORM get_data.

*........to process the data using loop and read......
  PERFORM process_data.

*........to display the data by calling SF...........
  PERFORM display_data.

ENDFORM.

FORM process_data .

  SORT lt_ekpo_itms BY ebelp.

  SORT lt_eslh_itm BY packno DESCENDING.

  DELETE ADJACENT DUPLICATES FROM lt_eslh_itm COMPARING ebelp.

  SORT lt_eslh_itm BY ebelp.

  LOOP AT lt_ekpo_itms ASSIGNING FIELD-SYMBOL(<fs_ekpo>).
    READ TABLE lt_eslh_itm ASSIGNING FIELD-SYMBOL(<fs_eslh>) WITH KEY ebelp = <fs_ekpo>-ebelp BINARY SEARCH.
    IF sy-subrc = 0.
      <fs_eslh>-sr_no = sy-tabix.
      <fs_eslh>-txz01 = <fs_ekpo>-txz01.
      <fs_eslh>-menge = <fs_ekpo>-menge.
      <fs_eslh>-meins = <fs_ekpo>-meins.
      <fs_eslh>-mwskz = <fs_ekpo>-mwskz.
      <fs_eslh>-netwr = <fs_ekpo>-netwr.
    ENDIF.
  ENDLOOP.

  SORT lt_gst BY ebelp.
  SORT lt_konp BY knumh kschl.
  SORT lt_a003 BY mwskz.

  LOOP AT lt_ekpo_itms INTO DATA(lwa_ekpo1) ##INTO_OK.
    ls_netwr-ebeln = lwa_ekpo1-ebeln.
    ls_netwr-netwr = lwa_ekpo1-netwr.
    COLLECT ls_netwr INTO lt_netwr.

    READ TABLE lt_a003 INTO DATA(lwa_a003) WITH KEY mwskz = lwa_ekpo1-mwskz BINARY SEARCH  ##NEEDED.
    CLEAR lv_index.
    lv_index = sy-tabix.
    IF sy-subrc = 0.
      LOOP AT lt_a003 INTO DATA(ls_a003) FROM lv_index.
        IF lwa_ekpo1-mwskz <> ls_a003-mwskz.
          EXIT.
        ELSE.
          READ TABLE lt_konp INTO ls_konp WITH KEY knumh = ls_a003-knumh kschl = ls_a003-kschl BINARY SEARCH  ##NEEDED.
          IF sy-subrc = 0.
            CASE ls_a003-kschl.
              WHEN lc_jiig.
                lwa_gst-igst_per =  ls_konp-kbetr / 10.
                lwa_gst-igst = lwa_ekpo1-netpr * ( ls_konp-kbetr / 1000 ).
              WHEN lc_jicg.
                lwa_gst-cgst_per =  ls_konp-kbetr / 10.
                lwa_gst-cgst = lwa_ekpo1-netpr * ( ls_konp-kbetr / 1000 ).
              WHEN lc_jisg.
                lwa_gst-sgst_per =  ls_konp-kbetr / 10.
                lwa_gst-sgst = lwa_ekpo1-netpr * ( ls_konp-kbetr / 1000 ).
            ENDCASE.
          ENDIF.
        ENDIF.
      ENDLOOP.
    ENDIF.

    lwa_gst-ebeln = lwa_ekpo1-ebeln.
    lwa_gst-ebelp = lwa_ekpo1-ebelp.

    APPEND lwa_gst TO lt_gst.
    CLEAR: lwa_gst,ls_konp,ls_a003,lwa_ekpo1.
  ENDLOOP.

  CLEAR: lwa_gst.

  LOOP AT lt_gst INTO lwa_gst.
    READ TABLE lt_eslh_itm ASSIGNING <fs_eslh> WITH KEY ebelp = lwa_gst-ebelp BINARY SEARCH.
    IF sy-subrc = 0.
      <fs_eslh>-igst = lwa_gst-igst.
      <fs_eslh>-sgst = lwa_gst-sgst.
      <fs_eslh>-cgst = lwa_gst-cgst.
      <fs_eslh>-igst% = lwa_gst-igst_per.
      <fs_eslh>-sgst% = lwa_gst-sgst_per.
      <fs_eslh>-cgst% = lwa_gst-cgst_per.
    ENDIF.

    ls_collec-ebeln = lwa_gst-ebeln.
    ls_collec-igst = lwa_gst-igst.
    ls_collec-cgst = lwa_gst-cgst.
    ls_collec-sgst = lwa_gst-sgst.

    COLLECT ls_collec INTO lt_collec.
    CLEAR: ls_collec, lwa_gst.
  ENDLOOP.

  READ TABLE lt_collec INTO ls_collec INDEX 1.
  READ TABLE lt_netwr INTO ls_netwr INDEX 1.

  gs_print_data-valueofwrk = ls_netwr-netwr.
  gs_print_data-igst_sum = ls_collec-igst.
  gs_print_data-sgst_sum = ls_collec-sgst.
  gs_print_data-cgst_sum = ls_collec-cgst.

  CLEAR: ls_collec,ls_netwr,lt_netwr[],lt_collec[].

ENDFORM.

    MESSAGE TEXT-000 TYPE 'E'.
  ENDIF.

  IF s_gp_no IS NOT INITIAL.
    SELECT zgp_no FROM zmm_gph
    INTO TABLE @DATA(lt_check_gpno)
    WHERE zgp_no IN @s_gp_no.

    IF lt_check_gpno IS INITIAL .
      MESSAGE TEXT-002 TYPE 'E'.
      EXIT.
    ENDIF.
  ENDIF.

*--> selcting record from database based on input screen.
START-OF-SELECTION.

  "--> check and validate the Returnable Gatepass
  SELECT zgp_no,zgp_ty,gp_odt
  FROM zmm_gph
  INTO TABLE @lt_gpno
  WHERE zgp_no IN @s_gp_no AND
        zgp_ty EQ 'R' AND
        gp_odt IN @s_gp_odt.

  IF lt_gpno IS NOT INITIAL.
    PERFORM get_data.                    "--> Get Gate pass Necessary Details.
    PERFORM process_data.                "--> Prepare final Gate Pass Result set.
    PERFORM display_data.                "--> Display the Report in ALV format.
  ELSE.

    "--> Raise message: if no Returnable Gate Pass Record present in the date range.
    IF s_gp_odt IS NOT INITIAL.
      SELECT zgp_no,zgp_ty,gp_odt
      FROM zmm_gph
      INTO TABLE @DATA(lt_gpdat)
      WHERE zgp_ty EQ 'R' AND
            gp_odt IN @s_gp_odt.

      IF lt_gpdat IS INITIAL.
        MESSAGE TEXT-003 TYPE 'E' DISPLAY LIKE 'I'.
      ENDIF.

      "--> Raise message: if no Returnable Gate Pass Record present in the given Gatepass number and date range.
    ELSEIF s_gp_no IS NOT INITIAL AND s_gp_odt IS NOT INITIAL.
      SELECT zgp_no,zgp_ty,gp_odt
      FROM zmm_gph
      INTO TABLE @DATA(lt_gpno_date)
      WHERE zgp_no IN @s_gp_no AND
            zgp_ty EQ 'R' AND
            gp_odt IN @s_gp_odt.

      IF lt_gpno_date IS INITIAL.
        MESSAGE TEXT-004 TYPE 'E' DISPLAY LIKE 'I'.
      ENDIF.

      "--> Raise message: if no Returnable Gate Pass Record Available for the given gate pass Number .
    ELSE.
      MESSAGE TEXT-005 TYPE 'E' DISPLAY LIKE 'I'.
    ENDIF.
  ENDIF.

FORM process_data .
  "--> Prepare Item final Result.
  PERFORM process_item_result.

  "--> Fetch all the availbale user details.
  SELECT  bname,name_textc
  FROM user_addr INTO TABLE @DATA(lt_user).             "#EC CI_NOWHERE

  "--> Preparing the final result.
  IF  lt_rgp_header IS NOT INITIAL.
    LOOP AT lt_rgp_header INTO DATA(ls_header).
      ls_gp_result-zgp_no        = ls_header-zgp_no.
      ls_gp_result-zgp_ty        = ls_header-zgp_ty.
      IF ls_header-zgp_ty EQ 'R' .
        ls_gp_result-zgp_ty_t = 'R-Returnable Gate Pass'.
      ENDIF.
      ls_gp_result-Vehicle_No    = ls_header-vehicle_no.
      ls_gp_result-gp_odt        = ls_header-gp_odt.
      ls_gp_result-cons_nam      = ls_header-cons_nam.
      ls_gp_result-tele          = ls_header-tele.
      ls_gp_result-name1         = ls_header-name1.
      ls_gp_result-crnam         = ls_header-crnam.
      ls_gp_result-crdat         = ls_header-crdat.
      ls_gp_result-crtim         = ls_header-crtim.
      ls_gp_result-reason        = ls_header-reason.
      ls_gp_result-location      = ls_header-city.
      ls_gp_result-ref_doc       = ls_header-ref_doc.
      ls_gp_result-eway_no       = ls_header-eway.
      ls_gp_result-gst_no_vendor = ls_header-stcd3.
      ls_gp_result-return_updator = ls_header-aenam.

      "--> Fetch updator name.
      READ TABLE lt_user INTO DATA(ls_updator) WITH  KEY bname = ls_header-crnam.
      IF ls_updator IS NOT INITIAL.
        ls_gp_result-updator = |{ ls_updator-name_textc } ({ ls_header-crnam })| .
      ENDIF.

      "--> Fetch Return updator name.
      READ TABLE lt_user INTO DATA(ls_return_updator) WITH  KEY bname = ls_header-aenam.
      IF ls_return_updator IS NOT INITIAL.
        ls_gp_result-return_updator_name = |{ ls_return_updator-name_textc } ({ ls_header-aenam })| .
      ENDIF.

      "--> fetching approval detail.
      READ TABLE lt_gpa INTO DATA(wa_gpa) WITH KEY zgp_no = ls_header-zgp_no BINARY SEARCH.
      IF sy-subrc EQ 0.
        ls_gp_result-susid             = wa_gpa-susid.
        ls_gp_result-app_name_text     = |{ wa_gpa-name_text } ({ wa_gpa-susid })|.
        ls_gp_result-app_stat          = wa_gpa-app_stat.
        ls_gp_result-app_stat_text     = wa_gpa-app_stat_t.
        ls_gp_result-app_lvl           = wa_gpa-app_lvl.
        ls_gp_result-app_aedat         = wa_gpa-aedat.
        ls_gp_result-app_aezet         = wa_gpa-aezet.
      ENDIF.

      "--> Fetch Item Details.
      LOOP AT lt_item_final INTO DATA(ls_item).
        IF ls_item-rgpno = ls_header-zgp_no.
          ls_gp_result-item           = ls_item-item.
          ls_gp_result-matnr          = ls_item-matnr.
          ls_gp_result-maktx          = ls_item-maktx.
          ls_gp_result-mtart          = ls_item-mtart.
          ls_gp_result-meins          = ls_item-meins.
          ls_gp_result-menge          = ls_item-menge.
          ls_gp_result-reamrks        = ls_item-remarks.
          ls_gp_result-steuc          = ls_item-steuc.
          ls_gp_result-netwr          = ls_item-netwr.
          ls_gp_result-gst_val        = ls_item-gst_val.
          ls_gp_result-gtotal         = ls_item-gtotal.
          ls_gp_result-return_qty     = ls_item-recvng_qty.
          ls_gp_result-pending_qty    = ls_item-pending_qty.
          ls_gp_result-return_date    = ls_item-gp_indt.
          ls_gp_result-return_time    = ls_item-gp_intm.

          APPEND ls_gp_result TO lt_gp_result.
        ENDIF.
        CLEAR: ls_item,ls_gp_result-item ,ls_gp_result-matnr,
              ls_gp_result-maktx,ls_gp_result-meins,ls_gp_result-menge,
              ls_gp_result-reamrks,ls_gp_result-return_qty ,ls_gp_result-pending_qty ,
              ls_gp_result-return_date,ls_gp_result-return_time .
      ENDLOOP.
      CLEAR: ls_header,wa_gpa.
    ENDLOOP.
  ENDIF.

  SORT lt_gp_result BY zgp_no item return_date.
ENDFORM.

FORM get_rgp_approver_data .
  "--> Fetching all gate pass approval status details.
  SELECT domname,ddlanguage,valpos,domvalue_l,ddtext
  FROM dd07t INTO TABLE @DATA(lt_aprvl_status)
  WHERE domname    = 'ZDMAPPSTAT' AND
        ddlanguage = @sy-langu.

  "--> Fetch all the availbale user details.
  SELECT  bname,name_textc
  FROM user_addr
  INTO TABLE @DATA(lt_user).                            "#EC CI_NOWHERE

  IF lt_rgp_header IS NOT INITIAL.
    SELECT zgp_no,rel_strg,app_lvl,susid,app_stat,aedat,aezet
    FROM zmm_gpa
    FOR ALL ENTRIES IN @lt_rgp_header
    WHERE zgp_no  = @lt_rgp_header-zgp_no
    INTO TABLE @DATA(lt_gpa_temp).
  ENDIF.

  "--> sorting the "Gate Pass Release Strategy" table data in decending
  "    where the first index record repesent the updated approval status
  "    and delete the other index record by comparing the gate pass number.
  SORT lt_gpa_temp DESCENDING BY zgp_no app_lvl.
  DELETE ADJACENT DUPLICATES FROM lt_gpa_temp COMPARING zgp_no.

  IF  lt_gpa_temp IS NOT INITIAL.
    SORT lt_gpa_temp BY zgp_no.

    LOOP AT lt_rgp_header INTO DATA(ls_header).
      READ TABLE lt_gpa_temp INTO DATA(ls_gpa_temp) WITH KEY zgp_no = ls_header-zgp_no BINARY SEARCH.
      IF ls_gpa_temp IS NOT INITIAL .

        ls_gpa-zgp_no   = ls_header-zgp_no.
        ls_gpa-zgp_ty   = ls_header-zgp_ty.
        ls_gpa-rel_strg = ls_gpa_temp-rel_strg.
        ls_gpa-app_lvl  = ls_gpa_temp-app_lvl.
        ls_gpa-susid    = ls_gpa_temp-susid.
        ls_gpa-app_stat = ls_gpa_temp-app_stat.
        ls_gpa-aedat    = ls_gpa_temp-aedat.
        ls_gpa-aezet    = ls_gpa_temp-aezet.

        "--> Read Approver name.
        READ TABLE lt_USER INTO DATA(ls_USER) WITH  KEY bname = ls_gpa-susid.
        IF ls_USER IS NOT INITIAL.
          ls_gpa-name_text = ls_user-name_textc .
        ENDIF.

        "--> Read approval status text.
        READ TABLE lt_aprvl_status INTO DATA(ls_aprvl_status) WITH  KEY domvalue_l = ls_gpa_temp-app_stat.
        IF ls_aprvl_status IS NOT INITIAL.
          ls_gpa-app_stat_t = |{ ls_gpa_temp-app_stat } ({ ls_aprvl_status-ddtext })|.
        ENDIF.

        APPEND ls_gpa TO lt_gpa.
      ENDIF.
      CLEAR : ls_gpa_temp,ls_header,ls_gpa,ls_aprvl_status.
    ENDLOOP.
  ENDIF.

  SORT lt_gpa BY zgp_no.

ENDFORM.

FORM get_rgp_item_data .

           zmm_gpi~maktx,zmm_gpi~meins,zmm_gpi~menge,

FORM get_rgp_reciving_data .
  IF lt_rgp_header IS NOT INITIAL.

    " 1--> Fetching Gate Pass RGP Receiving Header details as per Gate pass No .
    SELECT rgprno,rgpno,gp_indt,gp_intm
    FROM zmm_gprh
    FOR ALL ENTRIES IN @lt_rgp_header
    WHERE rgpno = @lt_rgp_header-zgp_no
    INTO TABLE @DATA(lt_gprh).

    SORT lt_gprh BY rgprno gp_indt.

    " 2--> Fetching all the possible Gate Pass RGP Receiving Item details  .
    IF lt_gprh IS NOT INITIAL.
      SELECT rgprno,item,recvng_qty,meins
      FROM zmm_gpri
      FOR ALL ENTRIES IN @lt_gprh
      WHERE rgprno = @lt_gprh-rgprno
      INTO TABLE @DATA(lt_gpri).
    ENDIF.

    DELETE lt_gpri WHERE recvng_qty LE 0.
    SORT lt_gpri BY rgprno item.

    " 3--> Preparing the reciving item details.
    LOOP AT lt_rgp_header INTO DATA(ls_rgp_h).
      LOOP AT lt_gprh INTO DATA(ls_gprh) WHERE rgpno EQ ls_rgp_h-zgp_no.
        LOOP AT lt_rgp_item INTO DATA(ls_rgp_i) WHERE zgp_no EQ ls_rgp_h-zgp_no.

          READ TABLE lt_gpri INTO DATA(ls_gpri) WITH KEY rgprno = ls_gprh-rgprno
                                                         item   = ls_rgp_i-item BINARY SEARCH.

          IF sy-subrc EQ 0.
            ls_reciving-rgpno      = ls_rgp_h-zgp_no.
            ls_reciving-rgprno     = ls_gpri-rgprno.
            ls_reciving-item       = ls_gpri-item.
            ls_reciving-recvng_qty = ls_gpri-recvng_qty.
            ls_reciving-meins      = ls_gpri-meins.
            ls_reciving-gp_indt    = ls_gprh-gp_indt.
            ls_reciving-gp_intm    = ls_gprh-gp_intm.

            APPEND ls_reciving TO lt_reciving.
          ENDIF.
          CLEAR: ls_rgp_i,ls_gpri,ls_reciving.
        ENDLOOP.
        CLEAR ls_gprh.
      ENDLOOP.
      CLEAR : ls_rgp_h.
    ENDLOOP.
  ENDIF.

  SORT lt_reciving BY rgpno rgprno item gp_indt gp_intm.
ENDFORM.

FORM process_item_result .

  "--> Local variable.
  DATA: lv_pending_qty TYPE zderecvngqty.

  IF  lt_rgp_item IS NOT INITIAL.

    LOOP AT lt_rgp_item INTO DATA(ls_item).
      ls_final_item-rgpno   = ls_item-zgp_no.
      ls_final_item-item    = ls_item-item.
      ls_final_item-matnr   = ls_item-matnr.
      ls_final_item-maktx   = ls_item-maktx.
      ls_final_item-mtart   = ls_item-mtart.
      ls_final_item-menge   = ls_item-menge.
      ls_final_item-meins   = ls_item-meins.
      ls_final_item-remarks = ls_item-remarks.
      ls_final_item-steuc   = ls_item-steuc.
      ls_final_item-netwr   = ls_item-netwr.
      ls_final_item-gst_val = ls_item-gst_val.
      ls_final_item-gtotal  = ls_item-gtotal.

      IF line_exists( lt_reciving[ rgpno = ls_item-zgp_no item = ls_item-item ] ).
        LOOP AT lt_reciving INTO DATA(ls_reciving).
          IF ls_reciving-rgpno = ls_item-zgp_no AND ls_reciving-item  = ls_item-item.

            ls_final_item-rgprno     = ls_reciving-rgprno.
            ls_final_item-recvng_qty = ls_reciving-recvng_qty.
            ls_final_item-gp_indt    = ls_reciving-gp_indt.
            ls_final_item-gp_intm    = ls_reciving-gp_intm.

            "--> calculating pending quantity.
            IF lv_pending_qty IS INITIAL.
              lv_pending_qty = ( ls_item-menge - ls_reciving-recvng_qty ).
            ELSE.
              lv_pending_qty = ( lv_pending_qty - ls_reciving-recvng_qty ).
            ENDIF.
            ls_final_item-pending_qty    = lv_pending_qty.

            "--> Insert record to internal table.
            APPEND ls_final_item TO lt_item_final .
          ENDIF.

          CLEAR: ls_reciving,ls_final_item-rgprno,ls_final_item-recvng_qty,
                 ls_final_item-gp_indt ,ls_final_item-gp_intm,ls_final_item-pending_qty.
        ENDLOOP.
      ELSE.
        APPEND ls_final_item TO lt_item_final.
      ENDIF.
      CLEAR: ls_item,lv_pending_qty.
    ENDLOOP.
  ENDIF.

  SORT lt_item_final BY rgpno item gp_indt gp_indt.
ENDFORM.

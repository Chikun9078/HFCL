          ix_screen      = ent_screen
        IMPORTING

          doc            = l_doc
        CHANGING

    ELSE.

    ENDIF.

        ix_screen      = ent_screen
      IMPORTING
        ex_retco       = ent_retco

    ELSE.

      ls_control_parameters-no_dialog  = 'X'.
      ls_control_parameters-getotf = 'X'.

      PERFORM fget_data.

    ENDIF.
  ENDIF.

ENDFORM.

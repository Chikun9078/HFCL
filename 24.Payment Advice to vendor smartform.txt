FORM get_header_final_details .
  IF lt_bkpf IS NOT INITIAL.
    "--> Fetch Vendor Details.
    SELECT bseg~bukrs,
           bseg~belnr,
           bseg~gjahr,
           bseg~koart,
           bseg~lifnr,
           lfa1~adrnr,
           adrc~name1,
           adrc~street,
           adrc~str_suppl1,
           adrc~str_suppl2,
           adrc~str_suppl3,
           adrc~post_code1,
           adrc~city1,
           adrc~region,
           adrc~country,
           t005u~bezei,
           lfbk~bankl,
           lfbk~bankn
    FROM bseg
    INNER JOIN lfa1 ON lfa1~lifnr EQ bseg~lifnr
    LEFT OUTER JOIN adrc ON adrc~addrnumber EQ lfa1~adrnr
    LEFT OUTER JOIN t005u ON t005u~bland EQ adrc~region AND
                             t005u~land1 EQ adrc~country AND
                             t005u~spras EQ @sy-langu
    LEFT OUTER JOIN lfbk ON lfbk~lifnr EQ bseg~lifnr AND "#EC CI_BUFFJOIN
                            lfbk~banks EQ adrc~country
    INTO TABLE @DATA(lt_vendor)
    FOR ALL ENTRIES IN @lt_bkpf
    WHERE bseg~belnr EQ @lt_bkpf-belnr AND
          bseg~bukrs EQ @lt_bkpf-bukrs AND
          bseg~gjahr EQ @lt_bkpf-gjahr AND
          bseg~koart EQ 'K'.
    SORT lt_vendor BY belnr.

    "--> Get Payment Amount Detail.
    ""--> from BSIK table.
    SELECT bukrs,gjahr,augbl,belnr,buzei,blart,shkzg,dmbtr,waers,rebzg
    FROM bsik_view INTO TABLE @DATA(lt_bsik)
    FOR ALL ENTRIES IN @lt_bkpf
    WHERE bukrs EQ @lt_bkpf-bukrs AND
          belnr EQ @lt_bkpf-belnr AND
          gjahr EQ @lt_bkpf-gjahr AND
          blart EQ 'KZ'.
    IF sy-subrc EQ 0.
      SORT lt_bsik BY belnr shkzg.
    ENDIF.

    IF lt_bsik IS INITIAL.
      ""--> From BSAK Table pass BKPF-Belnr in AUGBL.
      SELECT bukrs,gjahr,augbl,belnr,buzei,blart,shkzg,dmbtr,waers,rebzg
      FROM bsak_view INTO TABLE @DATA(lt_bsak_AUGBL)
      FOR ALL ENTRIES IN @lt_bkpf
      WHERE bukrs EQ @lt_bkpf-bukrs AND
            augbl EQ @lt_bkpf-belnr AND
            gjahr EQ @lt_bkpf-gjahr AND
            blart EQ 'KZ'.
      IF sy-subrc EQ 0.
        SORT lt_bsak_AUGBL BY augbl shkzg.
      ENDIF.

      IF lt_bsak_augbl IS INITIAL.
        ""--> From BSAK Table pass BKPF-Belnr in BSAK-BELNR.
        SELECT bukrs,gjahr,augbl,belnr,blart,buzei,shkzg,dmbtr,waers,rebzg
        FROM bsak_view INTO TABLE @DATA(lt_bsak_BELNR)
        FOR ALL ENTRIES IN @lt_bkpf
        WHERE bukrs EQ @lt_bkpf-bukrs AND
              belnr EQ @lt_bkpf-belnr AND
              gjahr EQ @lt_bkpf-gjahr AND
              blart EQ 'KZ'.
        IF sy-subrc EQ 0.
          SORT lt_bsak_belnr BY belnr shkzg.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

  "--> Prepare The final header Detail Result.
  LOOP AT lt_bkpf INTO DATA(ls_bkpf).
    ls_final_header-belnr = ls_bkpf-belnr.
    ls_final_header-bukrs = ls_bkpf-bukrs.
    ls_final_header-gjahr = ls_bkpf-gjahr.
    ls_final_header-budat = ls_bkpf-budat.
    ls_final_header-form_titel = 'Payment Advice'.

    READ TABLE lt_vendor INTO DATA(ls_vendor) WITH KEY belnr = ls_bkpf-belnr.
    IF sy-subrc EQ 0.
      ls_final_header-lifnr      = ls_vendor-lifnr.
      ls_final_header-lifnr_name = ls_vendor-name1.
      ls_final_header-street     = ls_vendor-street.
      ls_final_header-strspp1    = ls_vendor-str_suppl1.
      ls_final_header-strspp2    = ls_vendor-str_suppl2.
      ls_final_header-strspp3    = ls_vendor-str_suppl3.
      ls_final_header-region     = ls_vendor-region.
      ls_final_header-regio_desc = ls_vendor-bezei.
      ls_final_header-post_code  = ls_vendor-post_code1.
      ls_final_header-country    = ls_vendor-country.
      ls_final_header-city       = ls_vendor-city1.
      ls_final_header-lifnr_address = |{ ls_vendor-street },{ ls_vendor-str_suppl1 },{ ls_vendor-str_suppl2 },{ ls_vendor-str_suppl3 },{ ls_vendor-city1 },{ ls_vendor-bezei },{ ls_vendor-post_code1 }|.
      ls_final_header-bank_key      = ls_vendor-bankl.
      ls_final_header-bank_acct     = ls_vendor-bankn.
    ENDIF.

*   "--> get payment amount details of each doument.
    READ TABLE lt_bsik INTO DATA(ls_bsik) WITH KEY belnr = ls_bkpf-belnr.
    IF sy-subrc EQ 0.
      DATA(lv_index) = sy-tabix.
      ls_final_header-currency_key = ls_bsik-waers.
      ls_final_header-rebzg        = ls_bsik-rebzg.
      CLEAR:ls_bsik.

      LOOP AT lt_bsik INTO  ls_bsik FROM lv_index.
        IF ls_bsik-belnr EQ ls_bkpf-belnr.
          IF ls_bsik-shkzg EQ 'S'.
            lv_dmbtr_s = lv_dmbtr_s + ls_bsik-dmbtr.
          ELSEIF ls_bsik-shkzg EQ 'H'.
            lv_dmbtr_H = lv_dmbtr_H + ls_bsik-dmbtr.
          ENDIF.
        ELSE.
          EXIT.
        ENDIF.
        CLEAR:ls_bsik.
      ENDLOOP.
    ENDIF.

    CLEAR:lv_index.
    READ TABLE lt_bsak_augbl INTO DATA(ls_bsak) WITH KEY augbl = ls_bkpf-belnr.
    IF sy-subrc EQ 0.
      lv_index = sy-tabix.
      ls_final_header-currency_key = ls_bsak-waers.
      ls_final_header-rebzg        = ls_bsik-rebzg.
      CLEAR: ls_bsak.
      LOOP AT lt_bsak_augbl INTO ls_bsak FROM lv_index.
        IF ls_bsak-augbl EQ ls_bkpf-belnr.
          IF ls_bsak-shkzg EQ 'S'.
            lv_dmbtr_s = lv_dmbtr_s + ls_bsak-dmbtr.
          ELSEIF ls_bsak-shkzg EQ 'H'.
            lv_dmbtr_H = lv_dmbtr_H + ls_bsak-dmbtr.
          ENDIF.
        ELSE.
          EXIT.
        ENDIF.
        CLEAR:ls_bsak.
      ENDLOOP.
    ENDIF.

    CLEAR:lv_index,ls_bsak.
    READ TABLE lt_bsak_belnr INTO ls_bsak WITH KEY belnr = ls_bkpf-belnr.
    IF sy-subrc EQ 0.
      lv_index = sy-tabix.
      ls_final_header-currency_key = ls_bsak-waers.
      ls_final_header-rebzg        = ls_bsik-rebzg.
      CLEAR: ls_bsak.
      LOOP AT lt_bsak_belnr INTO ls_bsak FROM lv_index.
        IF ls_bsak-belnr EQ ls_bkpf-belnr.
          IF ls_bsak-shkzg EQ 'S'.
            lv_dmbtr_s = lv_dmbtr_s + ls_bsak-dmbtr.
          ELSEIF ls_bsak-shkzg EQ 'H'.
            lv_dmbtr_H = lv_dmbtr_H + ls_bsak-dmbtr.
          ENDIF.
        ELSE.
          EXIT.
        ENDIF.
        CLEAR:ls_bsak.
      ENDLOOP.
    ENDIF.

    ls_final_header-payment_amount       = ( lv_dmbtr_S - lv_dmbtr_h ).
    "--> Company Code Address.
    IF ls_bkpf-bukrs EQ '1100'.
      ls_final_header-bukrs_address = |8,Commercial Complex,Masjid Moth,Greater Kailash II,New Delhi,110048|.
      ls_final_header-bukrs_name    = 'Himachal Futuristic Communications Ltd (HFCL)'.
    ELSEIF ls_bkpf-bukrs EQ '1200'.
      ls_final_header-bukrs_address = |G.S.T. Road Guindy,Chennai,600032|.
      ls_final_header-bukrs_name    = 'Hindustan Teleprinters Limited (HTL)'.
    ENDIF.

    APPEND ls_final_header TO lt_header_final.
    CLEAR:ls_final_header,lv_dmbtr_s,lv_dmbtr_h.
  ENDLOOP.
ENDFORM.

FORM get_item_final_details .

  IF lt_header_final IS NOT INITIAL.
    SELECT  bukrs,augbl,gjahr,belnr,xblnr,budat,blart
    FROM bsak_view INTO CORRESPONDING FIELDS OF TABLE @lt_bsak1
    FOR ALL ENTRIES IN @lt_header_final
    WHERE augbl =  @lt_header_final-belnr AND
          belnr NE @lt_header_final-belnr AND
          bukrs =  @lt_header_final-bukrs .
*      AND gjahr =  @lt_header_final-gjahr.
  ENDIF.
  SORT lt_bsak1 BY augbl belnr DESCENDING.

  IF lt_bsak1 IS NOT INITIAL.
    SELECT  bukrs,augbl,gjahr,belnr,xblnr,budat,blart,shkzg,dmbtr,qbshb,waers,umskz
    FROM bsik_view INTO TABLE @lt_bsak2
    FOR ALL ENTRIES IN @lt_bsak1
    WHERE belnr = @lt_bsak1-belnr AND
          bukrs = @lt_bsak1-bukrs AND
          gjahr = @lt_bsak1-gjahr.

    IF lt_bsak2 IS INITIAL.
      SELECT  bukrs,augbl,gjahr,belnr,xblnr,budat,blart,shkzg,dmbtr,qbshb,waers,umskz
      FROM bsak_view INTO TABLE @lt_bsak2
      FOR ALL ENTRIES IN @lt_bsak1
      WHERE belnr = @lt_bsak1-belnr AND
            bukrs = @lt_bsak1-bukrs AND
            gjahr = @lt_bsak1-gjahr.
    ENDIF.

    SORT lt_bsak2 BY belnr DESCENDING.

    "--> Get Tax details.
    SELECT bukrs,belnr,gjahr,buzei,hwbas,hwste,ktosl,shkzg
    FROM bset INTO TABLE @lt_bset
    FOR ALL ENTRIES IN @lt_bsak1
    WHERE belnr EQ @lt_bsak1-belnr AND
          gjahr EQ @lt_bsak1-gjahr AND
          bukrs EQ @lt_bsak1-bukrs.

    SORT lt_bset BY belnr DESCENDING.
  ENDIF.

  "--> Prepare Final Item Result.
  LOOP AT lt_header_final INTO DATA(ls_header).
    "--> Check for packaging Document case.
    IF ls_header-rebzg IS INITIAL.
      READ TABLE lt_bsak1 INTO DATA(temp_bsak1) WITH KEY augbl = ls_header-belnr.
      IF sy-subrc EQ 0.
        DATA(lv_index2) = sy-tabix.
      ELSE.
        EXIT.
      ENDIF.
      LOOP AT lt_bsak1 INTO DATA(ls_bsak1) FROM lv_index2.
        IF ls_bsak1-augbl EQ ls_header-belnr .
          ls_final_item-belnr = ls_bsak1-augbl.
          ls_final_item-bukrs = ls_bsak1-bukrs.
          ls_final_item-gjahr = ls_bsak1-gjahr.
          ls_final_item-xblnr = ls_bsak1-xblnr.
          ls_final_item-budat = ls_bsak1-budat.
          ls_final_item-belnr_inv = ls_bsak1-belnr.

          IF ls_bsak1-blart EQ 'KR' OR ls_bsak1-blart EQ 'KG' OR ls_bsak1-blart EQ 'RE'.

            "-->amount payable detail and Gst.
            READ TABLE lt_bset INTO DATA(ls_bset) WITH KEY belnr = ls_bsak1-belnr.
            IF sy-subrc EQ 0.
              ls_final_item-amount_payable = ls_bset-hwbas.
              DATA(lv_index3) = sy-tabix.
            ENDIF.
            CLEAR:lv_dmbtr_s,lv_dmbtr_h.
            LOOP AT lt_bset INTO ls_bset FROM lv_index3 WHERE hwste GT 0.
              IF ls_bset-belnr EQ ls_bsak1-belnr.
                IF ls_bset-shkzg EQ 'S'.
                  lv_dmbtr_s = lv_dmbtr_s + LS_bset-hwste.
                ELSEIF LS_bset-shkzg EQ 'H'.
                  lv_dmbtr_H = lv_dmbtr_H + LS_bset-hwste.
                ENDIF.
                CLEAR:LS_bset.
              ENDIF.
            ENDLOOP.
            ls_final_item-gst_amount = lv_dmbtr_s - lv_dmbtr_h.



*          LOOP AT lt_bSet INTO ls_bset FROM lv_index3.
*            IF ls_bset-belnr EQ ls_bsak1-belnr.
*              lv_gst = lv_gst + ls_bset-hwste .
*            ELSE.
*              EXIT.
*            ENDIF.
*          ENDLOOP.
*          ls_final_item-gst_amount = lv_gst.
*          CLEAR:lv_gst.

            "--> Get TDS Amount Detail.
            READ TABLE lt_bsak2 INTO DATA(ls_bsak2) WITH KEY belnr = ls_bsak1-belnr.
            IF sy-subrc EQ 0.
              ls_final_item-tds_amount = ls_bsak2-qbshb.
            ENDIF.

            "--> Get Net Paid Amount.
            ls_final_item-net_paid = ls_final_item-amount_payable + ls_final_item-gst_amount - ls_final_item-tds_amount - ls_final_item-payment_less - ls_final_item-shkzg_note.

          ELSEIF ls_bsak1-blart EQ 'IU'.
            "--> Get Payment Less Amount.
            READ TABLE lt_bsak2 INTO ls_bsak2 WITH KEY belnr = ls_bsak1-belnr.
            IF sy-subrc EQ 0.
              DATA(lv_index5) = sy-tabix.
            ENDIF.
            CLEAR: ls_bsak2,lv_dmbtr_h,lv_dmbtr_s.
            LOOP AT lt_bsak2 INTO ls_bsak2 FROM lv_index5.
              IF ls_bsak2-belnr EQ ls_bsak1-belnr.
                IF ls_bsak2-shkzg EQ 'S'.
                  lv_dmbtr_s = lv_dmbtr_s + ls_bsak2-dmbtr.
                ELSEIF ls_bsak2-shkzg EQ 'H'.
                  lv_dmbtr_h = lv_dmbtr_H + ls_bsak2-dmbtr.
                ENDIF.
              ELSE.
                EXIT.
              ENDIF.
              CLEAR:ls_bsak2.
            ENDLOOP.
            ls_final_item-amount_payable = lv_dmbtr_s - lv_dmbtr_h.

            "--> Get Net Paid Amount.
            ls_final_item-net_paid = ls_final_item-amount_payable + ls_final_item-gst_amount - ls_final_item-tds_amount - ls_final_item-payment_less - ls_final_item-shkzg_note.

          ELSE.
            "--> Get Payable Amount.
            CLEAR:lv_index5.
            READ TABLE lt_bsak2 INTO ls_bsak2 WITH KEY belnr = ls_bsak1-belnr.
            IF sy-subrc EQ 0.
              lv_index5 = sy-tabix.
            ENDIF.
            CLEAR: ls_bsak2,lv_dmbtr_h,lv_dmbtr_s.
            LOOP AT lt_bsak2 INTO ls_bsak2 FROM lv_index5.
              IF ls_bsak2-belnr EQ ls_bsak1-belnr.
                IF ls_bsak2-shkzg EQ 'S'.
                  lv_dmbtr_s = lv_dmbtr_s + ls_bsak2-dmbtr.
                ELSEIF ls_bsak2-shkzg EQ 'H'.
                  lv_dmbtr_h = lv_dmbtr_H + ls_bsak2-dmbtr.
                ENDIF.
              ELSE.
                EXIT.
              ENDIF.
              CLEAR:ls_bsak2.
            ENDLOOP.
            ls_final_item-payment_less = lv_dmbtr_s - lv_dmbtr_h.

            "--> Get Net Paid Amount.
            ls_final_item-net_paid = ls_final_item-amount_payable + ls_final_item-gst_amount - ls_final_item-tds_amount - ls_final_item-payment_less - ls_final_item-shkzg_note.
          ENDIF.
          APPEND ls_final_item TO lt_final_item.
          CLEAR: ls_final_item.
        ELSE.
          EXIT.
        ENDIF.
      ENDLOOP.
      CLEAR: ls_header,lv_index2.
    ELSE.
      "--> IF flow of logic comes to this Else bLock then, this payment is related to packahing and details will get from only BSIK_VIEW.
      SELECT bukrs,belnr,gjahr,xblnr,budat,blart,dmbtr,rebzg
      FROM bsik_view INTO TABLE @DATA(lt_packaging_bsik1)
      WHERE belnr EQ @ls_header-rebzg AND bukrs EQ @ls_header-bukrs.

      SELECT bukrs,belnr,gjahr,xblnr,budat,blart,dmbtr,rebzg
      FROM bsik_view INTO TABLE @DATA(lt_packaging_bsik2)
      WHERE rebzg EQ @ls_header-rebzg AND bukrs EQ @ls_header-bukrs.

      IF lt_packaging_bsik1 IS NOT INITIAL.
        "--> Get Gst Details.
        SELECT bukrs,belnr,gjahr,buzei,hwbas,hwste,ktosl,shkzg
        FROM bset INTO TABLE @DATA(lt_bset_p1)
        FOR ALL ENTRIES IN @lt_packaging_bsik1
        WHERE belnr EQ @lt_packaging_bsik1-belnr AND bukrs EQ @lt_packaging_bsik1-bukrs AND gjahr EQ @lt_packaging_bsik1-gjahr.

        "--> GET TDS/Withholding Tax Details.
        SELECT bukrs,belnr,gjahr,wt_qbshh
        FROM with_item INTO TABLE @DATA(lt_with_item_p1)
        FOR ALL ENTRIES IN @lt_packaging_bsik1
        WHERE belnr EQ @lt_packaging_bsik1-belnr AND bukrs EQ @lt_packaging_bsik1-bukrs AND gjahr EQ @lt_packaging_bsik1-gjahr.
      ENDIF.

      IF lt_packaging_bsik2 IS NOT INITIAL.
        "--> Get Gst Details.
        SELECT bukrs,belnr,gjahr,buzei,hwbas,hwste,ktosl,shkzg
        FROM bset INTO TABLE @DATA(lt_bset_p2)
        FOR ALL ENTRIES IN @lt_packaging_bsik2
        WHERE belnr EQ @lt_packaging_bsik2-belnr AND bukrs EQ @lt_packaging_bsik2-bukrs AND gjahr EQ @lt_packaging_bsik2-gjahr.

        "--> GET TDS/Withholding Tax Details.
        SELECT bukrs,belnr,gjahr,wt_qbshh
        FROM with_item INTO TABLE @DATA(lt_with_item_p2)
        FOR ALL ENTRIES IN @lt_packaging_bsik2
        WHERE belnr EQ @lt_packaging_bsik2-belnr AND bukrs EQ @lt_packaging_bsik2-bukrs AND gjahr EQ @lt_packaging_bsik2-gjahr.
      ENDIF.

      "--. Prepare item result.
      LOOP AT lt_packaging_bsik1 INTO DATA(ls_packaging_bsik1).
        ls_final_item-belnr = ls_header-belnr.
        ls_final_item-bukrs = ls_header-bukrs.
        ls_final_item-gjahr = ls_header-gjahr.
        ls_final_item-xblnr = ls_packaging_bsik1-xblnr.
        ls_final_item-budat = ls_packaging_bsik1-budat.
        ls_final_item-belnr_inv = ls_packaging_bsik1-belnr.


        "-->TDS AMOUNT.
        READ TABLE lt_with_item_p1 INTO DATA(wa_with_item) WITH KEY belnr = ls_packaging_bsik1-belnr bukrs = ls_packaging_bsik1-bukrs gjahr = ls_packaging_bsik1-gjahr.
        IF sy-subrc EQ 0.
          ls_final_item-tds_amount = wa_with_item-wt_qbshh.
        ENDIF.

        "--> Gst Amount.
        CLEAR:lv_dmbtr_s,lv_dmbtr_h.
        READ TABLE lt_bset_p1 INTO DATA(wa_bset) WITH KEY belnr = ls_packaging_bsik1-belnr bukrs = ls_packaging_bsik1-bukrs gjahr = ls_packaging_bsik1-gjahr.
        IF sy-subrc EQ 0.
          ls_final_item-amount_payable = wa_bset-hwbas.
          CLEAR: wa_bset.
          LOOP AT lt_bset_p1 INTO wa_bset.
            IF wa_bset-shkzg EQ 'S'.
              lv_dmbtr_s = lv_dmbtr_s + wa_bset-hwste.
            ELSEIF wa_bset-shkzg EQ 'H'.
              lv_dmbtr_H = lv_dmbtr_H + wa_bset-hwste.
            ENDIF.
            CLEAR:wa_bset.
          ENDLOOP.
          ls_final_item-gst_amount = lv_dmbtr_s - lv_dmbtr_h.

        ls_final_item-xblnr = ls_bsik-xblnr.
        ls_final_item-budat = ls_bsik-budat.
        IF ls_bsik-rebzg IS INITIAL.
          ls_final_item-belnr_inv = ls_bsik-belnr.
        ELSE.
          ls_final_item-belnr_inv = ls_bsik-rebzg.
        ENDIF.

        "--> Amount Payable.
        CLEAR:lv_dmbtr_s,lv_dmbtr_h.
        LOOP AT lt_bsik2 INTO  DATA(wa_bsik).
          IF wa_bsik-shkzg EQ 'S'.
            lv_dmbtr_s = lv_dmbtr_s + wa_bsik-dmbtr.
          ELSEIF wa_bsik-shkzg EQ 'H'.
            lv_dmbtr_H = lv_dmbtr_H + wa_bsik-dmbtr.

          CLEAR:wa_bsik.

        IF sy-subrc EQ 0.

        CLEAR:lv_dmbtr_s,lv_dmbtr_h,wa_bset.
        LOOP AT lt_bset INTO wa_bset.
          IF wa_bset-shkzg EQ 'S'.
            lv_dmbtr_s = lv_dmbtr_s + wa_bset-hwste.
          ELSEIF wa_bset-shkzg EQ 'H'.
            lv_dmbtr_H = lv_dmbtr_H + wa_bset-hwste.
          ENDIF.
          CLEAR:wa_bset.
        ENDLOOP.
        ls_final_item-gst_amount = lv_dmbtr_s - lv_dmbtr_h.

        "--> Get Net Paid Amount.
        ls_final_item-net_paid = ls_final_item-amount_payable + ls_final_item-gst_amount - ls_final_item-tds_amount - ls_final_item-payment_less - ls_final_item-shkzg_note.
        APPEND ls_final_item TO lt_final_item.
      ENDIF.

      "--> Check in Bsak for field Augbl.
      READ TABLE lt_bsak_augbl2 INTO DATA(ls_bsak) WITH KEY augbl = ls_header-belnr.
      IF sy-subrc EQ 0.
        ls_final_item-belnr = ls_bsak-augbl.
        ls_final_item-bukrs = ls_bsak-bukrs.
        ls_final_item-gjahr = ls_bsak-gjahr.
        ls_final_item-xblnr = ls_bsak-xblnr.
        ls_final_item-budat = ls_bsak-budat.
        IF ls_bsak-rebzg IS INITIAL.
          ls_final_item-belnr_inv = ls_bsak-belnr.
        ELSE.
          ls_final_item-belnr_inv = ls_bsak-rebzg.
        ENDIF.

        "--> Amount Payable.
        CLEAR:lv_dmbtr_s,lv_dmbtr_h.
        LOOP AT lt_bsak_augbl2 INTO  DATA(wa_bsak).
          IF wa_bsak-shkzg EQ 'S'.
            lv_dmbtr_s = lv_dmbtr_s + wa_bsak-dmbtr.
          ELSEIF wa_bsak-shkzg EQ 'H'.
            lv_dmbtr_H = lv_dmbtr_H + wa_bsak-dmbtr.
          ENDIF.
          CLEAR:wa_bsak.
        ENDLOOP.
        ls_final_item-amount_payable = lv_dmbtr_s - lv_dmbtr_h.

        "-->TDS AMOUNT.
        READ TABLE lt_with_item INTO wa_with_item WITH KEY belnr = ls_header-belnr bukrs = ls_header-bukrs gjahr = ls_header-gjahr.
        IF sy-subrc EQ 0.
          ls_final_item-tds_amount = wa_with_item-wt_qbshh.
        ENDIF.

        "--> Gst Amount.
        CLEAR:lv_dmbtr_s,lv_dmbtr_h.
        LOOP AT lt_bset INTO wa_bset.

            lv_dmbtr_s = lv_dmbtr_s + wa_bset-hwste.

        ENDLOOP.

        "--> Get Net Paid Amount.

      IF sy-subrc EQ 0.
        ls_final_item-belnr = ls_bsak-belnr.

FORM process_data .

  "--> Perform sorting of final result table.
  SORT lt_header_final BY belnr.
  SORT lt_final_item BY belnr belnr_inv DESCENDING.

  LOOP AT lt_header_final INTO DATA(ls_header).

          lv_payment_less_t,lv_shkzg_note_t,lv_net_paid_t.

      IF ls_item-belnr EQ ls_header-belnr.
        lv_amount_payable_t = lv_amount_payable_t + ls_item-amount_payable.
        lv_gst_amount_t     = lv_gst_amount_t     + ls_item-gst_amount.

        lv_shkzg_note_t     = lv_shkzg_note_t     + ls_item-shkzg_note.
      ELSE.
        EXIT.
      ENDIF.
      CLEAR: ls_item.
    ENDLOOP.

    ls_header-amount_payable_t = lv_amount_payable_t.
    ls_header-gst_amount_t     = lv_gst_amount_t.
    ls_header-tds_amount_t     = lv_tds_amount_t.
    ls_header-payment_less_t   = lv_payment_less_t.
    ls_header-shkzg_note_t     = lv_shkzg_note_t.

    MODIFY lt_header_final FROM ls_header.
    CLEAR:ls_header.
  ENDLOOP.
ENDFORM.

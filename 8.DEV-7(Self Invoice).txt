  PARAMETERS: p_bukrs TYPE bseg-bukrs OBLIGATORY,
              p_gjahr TYPE bseg-gjahr OBLIGATORY.
SELECTION-SCREEN END OF BLOCK b1.

*--> Validating the User Input.
AT SELECTION-SCREEN.

  "--> Validating Document Number and showing invalid document number .
  LOOP AT s_belnr ASSIGNING FIELD-SYMBOL(<fs_belnr>).
    SELECT SINGLE belnr FROM bkpf INTO @DATA(ls_belnr) WHERE belnr = @<fs_belnr>-low.
    IF sy-subrc NE 0.
      lv_count = lv_count + 1.
      IF lv_message IS INITIAL.
        lv_message = |{ <fs_belnr>-low }|.
      ELSE.
        lv_message = |{ lv_message },{ <fs_belnr>-low }|.
      ENDIF.
    ENDIF.
  ENDLOOP.

  IF lv_count > 0.
    lv_message = |{ lv_message }{ TEXT-002 }|.
    MESSAGE lv_message TYPE 'I' DISPLAY LIKE 'I'.
    LEAVE PROGRAM.
  ENDIF.

  "--> Validating Company Codeand showing invalid Company code.
  SELECT SINGLE werks FROM t001w INTO @DATA(ls_t001w) WHERE werks  = @p_bukrs .
  IF sy-subrc NE 0.
    MESSAGE TEXT-003 TYPE 'I' DISPLAY LIKE 'I'.
    LEAVE PROGRAM.
  ENDIF.

START-OF-SELECTION.
*--> Filtering Document Relataed to self invoice.
  SELECT bukrs,belnr,gjahr,buzei,koart,mwskz,bupla,lifnr
  FROM bseg INTO TABLE @DATA(lt_bseg_temp)
  WHERE belnr IN  @s_belnr AND
        gjahr = @p_gjahr AND
        bukrs = @p_bukrs AND
        koart = 'K'.

  "--> Fetching the Tax code Related to Self Invoive from ZSI_RMC_TAX(Self Invoice: RMS Tax Code) Datbase table .
  SELECT si_tax_code
  FROM zsi_rmc_tax INTO TABLE lt_tax.                "#EC CI_NOWHERE

  "--> Here if the Document's MWSKZ field Value lies in this Self Invoive tax code table,
  "    then only those document valid for Self Invoice print.
  LOOP AT  lt_bseg_temp INTO DATA(ls_data).
    IF line_exists( lt_tax[ si_tax_code = ls_data-mwskz ] ).
      APPEND ls_data TO lt_bseg_header.
    ENDIF.
  ENDLOOP.
  SORT lt_bseg_header BY belnr.

  IF lt_bseg_header IS NOT INITIAL .
    PERFORM get_data.                         " Subroutine To fetch all the Required Data.
    PERFORM process_data.                     " Subroutine to Prepare the final Result.
    PERFORM display_data.                     " Subroutine to Print the Smartform as ann output.

  ELSE.
    DESCRIBE TABLE lt_bseg_temp LINES DATA(count).
    IF lv_count LE 1.
      MESSAGE TEXT-004 TYPE 'E' DISPLAY LIKE 'I'.
    ELSE.
      MESSAGE TEXT-005 TYPE 'E' DISPLAY LIKE 'I'.
    ENDIF.
  ENDIF.

*& Form process_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM process_data .
  PERFORM get_header_result.             " Subroutine to prepare final Header Result.
  PERFORM get_item_result.               " Subroutine to prepare final Item Result.

  CLEAR : ls_header.
  LOOP AT lt_header_final INTO ls_header.

    CLEAR:lv_gt_cgst ,lv_gt_sgst,lv_gt_igst,lv_gt_ugst,
          lv_gt_total_amount,lv_gt_taxable_amount,lv_gt_total_amount_it,
          lv_gt_tax_amount,lv_gt_total_amount_it_word,lv_gt_tax_amount_word.

    "--> calculate the Grand total of total amount,taxable amount,cgst,sgst,igst,ugst,total tax,total amount(inc tax).
    LOOP AT  lt_item_final INTO DATA(ls_item).
      IF ls_item-ref_doc_no EQ ls_header-ref_doc_no.
        lv_gt_total_amount    = lv_gt_total_amount    + ls_item-total_amount.
        lv_gt_taxable_amount  = lv_gt_taxable_amount  + ls_item-taxable_amount.
        lv_discount           = lv_discount           + ls_item-discount.
        lv_gt_cgst            = lv_gt_cgst            + ls_item-cgst_amount.
        lv_gt_sgst            = lv_gt_sgst            + ls_item-sgst_amount.
        lv_gt_igst            = lv_gt_igst            + ls_item-igst_amount.
        lv_gt_ugst            = lv_gt_ugst            + ls_item-ugst_amount.
        lv_gt_total_amount_it = lv_gt_total_amount_it + ls_item-total_amount_it.
        lv_gt_tax_amount      = lv_gt_tax_amount      + ls_item-total_tax.
      ENDIF.
      CLEAR: ls_item.
    ENDLOOP.

    "--> convert Total Tax Tmount & Total Amount(Incl. Tax) in word.
    CALL FUNCTION 'HR_IN_CHG_INR_WRDS'
      EXPORTING
        amt_in_num         = CONV pc207-betrg( lv_gt_tax_amount )
      IMPORTING
        amt_in_words       = lv_gt_tax_amount_word
      EXCEPTIONS
        data_type_mismatch = 1
        OTHERS             = 2.
    IF sy-subrc <> 0.
      " Implement suitable error handling here
    ENDIF.

    TRANSLATE lv_gt_tax_amount_word TO LOWER CASE .
    SPLIT lv_gt_tax_amount_word AT space INTO TABLE DATA(itab) .
    LOOP AT itab ASSIGNING  FIELD-SYMBOL(<temp>) .
      lv_word = <temp> .
      TRANSLATE lv_word+0(1) TO UPPER CASE .
      CONCATENATE lv_result lv_word INTO lv_result SEPARATED BY space .
    ENDLOOP .
    CLEAR lv_gt_tax_amount_word.
    lv_gt_tax_amount_word = lv_result.
    CLEAR: lv_word,lv_result,itab.

    CALL FUNCTION 'HR_IN_CHG_INR_WRDS'
      EXPORTING
        amt_in_num         = CONV pc207-betrg( lv_gt_total_amount_it )
      IMPORTING
        amt_in_words       = lv_gt_total_amount_it_word
      EXCEPTIONS
        data_type_mismatch = 1
        OTHERS             = 2.
    IF sy-subrc <> 0.
      " Implement suitable error handling here
    ENDIF.

    TRANSLATE lv_gt_total_amount_it_word TO LOWER CASE .
    SPLIT lv_gt_total_amount_it_word AT space INTO TABLE itab .
    LOOP AT itab ASSIGNING <temp> .
      lv_word = <temp> .
      TRANSLATE lv_word+0(1) TO UPPER CASE .
      CONCATENATE  lv_result lv_word INTO lv_result SEPARATED BY space .
    ENDLOOP .
    CLEAR lv_gt_total_amount_it_word.
    lv_gt_total_amount_it_word = lv_result.
    CLEAR: lv_word,lv_result,itab.

    ls_header-gt_total_amount      = lv_gt_total_amount.
    ls_header-gt_discount          = lv_discount.
    ls_header-gt_taxable_amount    = lv_gt_taxable_amount.
    ls_header-gt_cgst_amount       = lv_gt_cgst.
    ls_header-gt_sgst_amount       = lv_gt_sgst.
    ls_header-gt_igst_amount       = lv_gt_igst.
    ls_header-gt_ugst_amount       = lv_gt_ugst.
    ls_header-gt_tl_amount_it      = lv_gt_total_amount_it.
    ls_header-gt_gst_amount        = lv_gt_tax_amount.
    ls_header-gt_tl_amount_it_word = COND #( WHEN lv_gt_total_amount_it IS INITIAL THEN 'NIL'
                                          ELSE |{ lv_gt_total_amount_it_word } only|
                                        ).
    ls_header-gt_gst_amount_word   = COND #( WHEN lv_gt_tax_amount IS INITIAL THEN 'NIL'
                                          ELSE |{ lv_gt_tax_amount_word } only|
                                        ).

    MODIFY lt_header_final FROM ls_header.
  ENDLOOP.

ENDFORM.

FORM get_gstin_supply_data .
  IF lt_bseg_header IS NOT INITIAL.
    SELECT j_1bbranch~bukrs,j_1bbranch~branch,j_1bbranch~adrnr,j_1bbranch~gstin,
           adrc~city1,adrc~region
    FROM j_1bbranch
    LEFT OUTER JOIN adrc ON j_1bbranch~adrnr = adrc~addrnumber                              "#EC CI_BUFFJOIN
    FOR ALL ENTRIES IN @lt_bseg_header
                    WHERE j_1bbranch~bukrs  = @lt_bseg_header-bukrs AND
                          j_1bbranch~branch = @lt_bseg_header-bupla
    INTO TABLE @lt_supply_gstin.
  ENDIF.
  SORT lt_supply_gstin BY bukrs branch.
ENDFORM.

*&---------------------------------------------------------------------*

FORM get_invoice_data .
  IF lt_bseg_header IS NOT INITIAL.
    SELECT bukrs,belnr,gjahr,budat,xblnr_alt
    FROM bkpf
    FOR ALL ENTRIES IN @lt_bseg_header
                    WHERE bukrs  = @lt_bseg_header-bukrs AND
                          belnr  = @lt_bseg_header-belnr AND
                          gjahr  = @lt_bseg_header-gjahr
    INTO TABLE @lt_invoice.
  ENDIF.
  SORT lt_invoice BY belnr.

ENDFORM.

FORM get_billto_data .
  IF lt_bseg_header IS NOT INITIAL.
    SELECT lfa1~lifnr,lfa1~land1,lfa1~adrnr,lfa1~regio,lfa1~j_1ipanno,lfa1~stcd3,
           t005u~bezei,
           adrc~name1,adrc~street,adrc~str_suppl1,adrc~str_suppl2,
           adrc~str_suppl3,adrc~post_code1
      FROM lfa1
      LEFT OUTER JOIN t005u ON lfa1~land1  = t005u~land1 AND
                               lfa1~regio  = t005u~bland AND
                               t005u~spras = @sy-langu
      LEFT OUTER JOIN adrc  ON lfa1~adrnr  = adrc~addrnumber                                     "#EC CI_BUFFJOIN
      FOR ALL ENTRIES IN @lt_bseg_header
      WHERE lfa1~lifnr = @lt_bseg_header-lifnr

*& Form get_item_result
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_item_result .
  DATA: lv_serial TYPE sy-index.

  IF lt_bseg_header IS NOT INITIAL.
    LOOP AT lt_bseg_header INTO DATA(ls_bseg).
      LOOP AT lt_hsn_amount INTO DATA(ls_hsn).
        IF  ls_hsn-belnr = ls_bseg-belnr AND ls_hsn-bukrs = ls_bseg-bukrs AND ls_hsn-gjahr = ls_bseg-gjahr.

          lv_serial          = lv_serial + 1.
          ls_item-sl_no      = lv_serial.                   " Serial Number.
          ls_item-ref_doc_no = ls_bseg-belnr.               " Reference Document Number.

          READ TABLE lt_srv_desc INTO DATA(ls_srv_desc) WITH KEY belnr = ls_hsn-belnr BINARY SEARCH.
          IF sy-subrc EQ 0.
            ls_item-srv_desc = ls_srv_desc-tdline.          " Description Of Goods/Service
          ENDIF.

          ls_item-hsn_sac        = ls_hsn-hsn_sac.            " HSN_SRC
*          ls_item-qty            =                             " Here quantity value fetched.Always the Qantity is 'NA' ,accoding to Funtional.
*          ls_item-unit           =                             " Here Unit value fetched.Always the unit value is ' '/Blank ,accoding to Funtional.
          ls_item-rate           = 100.                         " Here "Rate value" Always 100 ,accoding to Funtional.
          ls_item-total_amount   = ls_hsn-dmbtr.              " Total Amount.
*          ls_item-discount       =  '0.00'.                    " Here discount value Always 0.00, accoding to Funtional.
          ls_item-taxable_amount = ls_hsn-dmbtr.              " Taxable Amount.

          "/--> Fetch CGST Rate and Amount details.
          READ TABLE lt_gst_rate INTO DATA(ls_cgst_rate) WITH KEY belnr = ls_hsn-belnr
                                                                  kschl = 'JICG'.
          IF sy-subrc EQ 0.
            ls_item-cgst_rate = ls_cgst_rate-kbetr DIV 10.
          ENDIF.
          READ TABLE lt_gst_amount INTO DATA(ls_cgst_amount) WITH KEY belnr = ls_hsn-belnr
                                                        ktosl = 'JRC'.
          IF sy-subrc EQ 0.
            ls_item-cgst_amount = ls_cgst_amount-hwste.
          ENDIF.

          "/--> Fetch SGST Rate and Amount details.
          READ TABLE lt_gst_rate INTO DATA(ls_sgst_rate) WITH KEY belnr = ls_hsn-belnr
                                                                  kschl = 'JISG'.
          IF sy-subrc EQ 0.
            ls_item-sgst_rate = ls_sgst_rate-kbetr DIV 10.
          ENDIF.
          READ TABLE lt_gst_amount INTO DATA(ls_sgst_amount) WITH KEY belnr = ls_hsn-belnr
                                                                      ktosl = 'JRS'.
          IF sy-subrc EQ 0.
            ls_item-sgst_amount = ls_cgst_amount-hwste.
          ENDIF.
          CLEAR ls_sgst_amount.

          "--> Fetch IGST Rate and Amount details.
          READ TABLE lt_gst_rate INTO DATA(ls_igst_rate) WITH KEY belnr = ls_hsn-belnr
                                                                  kschl = 'JIIG'.
          IF sy-subrc EQ 0.
            ls_item-igst_rate = ls_igst_rate-kbetr DIV 10.
          ENDIF.
          READ TABLE lt_gst_amount INTO DATA(ls_igst_amount) WITH KEY belnr = ls_hsn-belnr
                                                                      ktosl = 'JRI'.
          IF sy-subrc EQ 0.
            ls_item-igst_amount = ls_igst_amount-hwste.
          ENDIF.

          "--> Fetch UGST Rate and Amount details.
          READ TABLE lt_gst_rate INTO DATA(ls_ugst_rate) WITH KEY belnr = ls_hsn-belnr
                                                                  kschl = 'JIUG'.
          IF sy-subrc EQ 0.
            ls_item-ugst_rate = ls_ugst_rate-kbetr DIV 10.
          ENDIF.
          READ TABLE lt_gst_amount INTO DATA(ls_ugst_amount) WITH KEY belnr = ls_hsn-belnr
                                                                      ktosl = 'JRU'.
          IF sy-subrc EQ 0.
            ls_item-ugst_amount = ls_ugst_amount-hwste.
          ENDIF.

          "/--> Prepare "Total Amount Including Tax" and "Total Tax" for the line item.
          ls_item-total_tax       = ( ls_item-cgst_amount + ls_item-sgst_amount + ls_item-igst_amount + ls_item-ugst_amount ).
          ls_item-total_amount_it = ( ls_item-total_amount + ls_item-total_tax ).

          " Insert Record into final item internal table.
          APPEND ls_item TO lt_item_final.

          CLEAR: ls_item,ls_srv_desc,ls_cgst_rate,ls_cgst_amount,
                 ls_sgst_rate,ls_sgst_amount,ls_igst_rate,ls_igst_amount,ls_ugst_rate,ls_ugst_amount.
        ENDIF.
        CLEAR: ls_hsn.
      ENDLOOP.
      CLEAR: lv_serial,ls_bseg.
    ENDLOOP.
  ENDIF.

  SORT lt_item_final BY sl_no ref_doc_no.
ENDFORM.	                                                                                                                                                                                                                                                               	______________________________

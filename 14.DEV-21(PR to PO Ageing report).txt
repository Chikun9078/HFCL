FORM process_data .
  "--> Preparing the final result set.
  LOOP AT lt_eban INTO DATA(ls_eban).
    ls_result-banfn        = ls_eban-banfn.
    ls_result-bnfpo        = ls_eban-bnfpo.
    ls_result-bsart        = ls_eban-bsart.
    ls_result-werks        = ls_eban-werks.
    ls_result-matnr        = ls_eban-matnr.
    ls_result-maktx        = ls_eban-maktx.
    ls_result-ekgrp_r      = |{ ls_eban-ekgrp }-{ ls_eban-eknam }|.
    ls_result-menge        = ls_eban-menge.
    ls_result-bumng        = ls_eban-bumng.
    ls_result-meins        = ls_eban-meins.
    ls_result-afnam        = ls_eban-afnam.
    ls_result-badat        = ls_eban-badat.
    ls_result-creationdate = ls_eban-creationdate.
    ls_result-frgdt        = ls_eban-frgdt.
    ls_result-ebeln        = ls_eban-ebeln.
    ls_result-ebelp        = ls_eban-ebelp.
    ls_result-bedat        = ls_eban-bedat.

    "--> Calulating the Number of days Taken.
    ls_result-days_taken_pocreation     = COND int2( WHEN ls_eban-bedat EQ ls_eban-frgdt THEN 0
                                                     WHEN ls_eban-bedat GT ls_eban-frgdt THEN ( ( ls_eban-bedat - ls_eban-frgdt ) + 1 )
                                                   ).

    ls_result-days_pending_ponotcreate  = COND int2( WHEN sy-datum EQ ls_eban-frgdt THEN 0
                                                     WHEN sy-datum GT ls_eban-frgdt THEN ( ( sy-datum - ls_eban-frgdt ) + 1 )
                                                   ).

    "--> Fetch the approver details like name,date,days taken for approval based on Approval Lvel.
    READ TABLE lt_approval INTO DATA(ls_approval) WITH KEY banfn = ls_eban-banfn BINARY SEARCH.
    IF  sy-subrc EQ 0 AND ls_approval IS NOT INITIAL.

      "--> first approval detail.
      IF ls_approval-frgc1 IS NOT INITIAL.
        ls_result-approval1_date = ls_approval-approval1_date.
        ls_result-approver1_id   = ls_approval-approver1_id.
        ls_result-approver1_name = ls_approval-approver1_name.
        ls_result-days_l1        = COND int2( WHEN ls_eban-creationdate EQ ls_approval-approval1_date THEN 0
                                              WHEN ls_approval-approval1_date GT ls_eban-creationdate THEN  ( ( ls_approval-approval1_date - ls_eban-creationdate ) + 1 )
                                            ).
      ENDIF.

      "--> second approval detail.
      IF ls_approval-frgc2 IS NOT INITIAL.
        ls_result-approval2_date = ls_approval-approval2_date.
        ls_result-approver2_id  = ls_approval-approver2_id.
        ls_result-approver2_name = ls_approval-approver2_name.
        ls_result-days_l2        = COND int2( WHEN ls_approval-approval2_date EQ ls_approval-approval1_date THEN 0
                                              WHEN ls_approval-approval2_date GT ls_approval-approval1_date THEN  ( ( ls_approval-approval2_date - ls_approval-approval1_date ) + 1 )
                                            ).
      ENDIF.

      "--> third approval detail.
      IF ls_approval-frgc3 IS NOT INITIAL.
        ls_result-approval3_date = ls_approval-approval3_date.
        ls_result-approver3_id   = ls_approval-approver3_id.
        ls_result-approver3_name = ls_approval-approver3_name.
        ls_result-days_l3        = COND int2( WHEN ls_approval-approval3_date EQ ls_approval-approval2_date THEN 0
                                              WHEN ls_approval-approval3_date GT ls_approval-approval2_date THEN  ( ( ls_approval-approval3_date - ls_approval-approval2_date ) + 1 )
                                            ).
      ENDIF.

      "--> Fourth approval detail.
      IF ls_approval-frgc4 IS NOT INITIAL.
        ls_result-approval4_date = ls_approval-approval4_date.
        ls_result-approver4_id   = ls_approval-approver4_id.
        ls_result-approver4_name = ls_approval-approver4_name.
        ls_result-days_l4        = COND int2( WHEN ls_approval-approval4_date EQ ls_approval-approval4_date THEN 0
                                              WHEN ls_approval-approval4_date GT ls_approval-approval4_date THEN  ( ( ls_approval-approval4_date - ls_approval-approval3_date ) + 1 )
                                            ).
      ENDIF.

      "--> fifth approval detail.
      IF ls_approval-frgc5 IS NOT INITIAL.
        ls_result-approval5_date = ls_approval-approval5_date.
        ls_result-approver5_id   = ls_approval-approver5_id.
        ls_result-approver5_name = ls_approval-approver5_name.
        ls_result-days_l5        = COND int2( WHEN ls_approval-approval5_date EQ ls_approval-approval4_date THEN 0
                                              WHEN ls_approval-approval5_date GT ls_approval-approval4_date THEN  ( ( ls_approval-approval5_date - ls_approval-approval4_date ) + 1 )
                                            ).
      ENDIF.

      "--> sixth approval detail.
      IF ls_approval-frgc6 IS NOT INITIAL.
        ls_result-approval6_date = ls_approval-approval6_date.
        ls_result-approver6_id   = ls_approval-approver6_id.
        ls_result-approver6_name = ls_approval-approver6_name.
        ls_result-days_l6        = COND int2( WHEN ls_approval-approval6_date EQ ls_approval-approval5_date THEN 0
                                              WHEN ls_approval-approval6_date GT ls_approval-approval5_date THEN  ( ( ls_approval-approval6_date - ls_approval-approval5_date ) + 1 )
                                            ).
      ENDIF.
    ENDIF.

    APPEND ls_result TO lt_result.
    CLEAR : ls_eban,ls_approval.
  ENDLOOP.

  SORT lt_result BY banfn bnfpo.
ENDFORM.

FORM get_eban_data .
  IF lt_valid IS NOT INITIAL .
    SELECT eban~banfn,eban~bnfpo,eban~bsart,eban~werks,eban~matnr,
           eban~ekgrp,eban~menge,eban~bumng,eban~meins,
           eban~afnam,eban~badat,eban~creationdate,eban~frgdt,
           eban~ebeln,eban~ebelp,eban~bedat,
           t024~eknam,
           makt~maktx
    FROM eban
    LEFT OUTER JOIN t024 ON eban~ekgrp EQ t024~ekgrp
    LEFT OUTER JOIN makt ON eban~matnr EQ makt~matnr AND "#EC CI_BUFFJOIN
                            makt~spras EQ @sy-langu
    INTO TABLE @lt_eban
    FOR ALL ENTRIES IN @lt_valid
    WHERE eban~banfn = @lt_valid-banfn.

  ENDIF.
  SORT lt_eban BY banfn bnfpo.
ENDFORM.

          ls_approval-approver1_id   = ls_t16fw1-objid.

        IF ls_t16fw4 IS NOT INITIAL.
          ls_approval-approver4_id   = ls_t16fw4-objid.
          ls_approval-approver4_name = ls_t16fw4-name_textc.
        ENDIF.

        READ TABLE lt_change INTO  DATA(ls_change4) WITH KEY objectid  = ls_valid-objectid  value_new = 'XXXX'.

          ls_approval-approval4_date = ls_change4-udate.

      IF ls_t16fs-frgc5 IS NOT INITIAL.
        ls_approval-frgc5 = ls_t16fs-frgc5.
        READ TABLE lt_t16fw INTO DATA(ls_t16fw5) WITH KEY frggr = ls_t16fs-frggr frgco = ls_t16fs-frgc5.
        IF ls_t16fw5 IS NOT INITIAL.
          ls_approval-approver5_id   = ls_t16fw5-objid.

        READ TABLE lt_change INTO  DATA(ls_change5) WITH KEY objectid  = ls_valid-objectid  value_new = 'XXXXX'.

          ls_approval-approval5_ind  = ls_change5-value_new.

          ls_approval-approver6_name = ls_t16fw6-name_textc.
        ENDIF.

        READ TABLE lt_change INTO  DATA(ls_change6) WITH KEY objectid  = ls_valid-objectid  value_new = 'XXXXXX'.
        IF ls_change6 IS NOT INITIAL.
          ls_approval-approval6_ind  = ls_change6-value_new.
          ls_approval-approval6_date = ls_change6-udate.
        ENDIF.
      ENDIF.
    ENDIF.

    APPEND ls_approval TO lt_approval.
    CLEAR: ls_approval,ls_valid,ls_t16fs,ls_t16fw1,ls_change1,
           ls_t16fw2,ls_change2,ls_t16fw3,ls_change3,
           ls_t16fw4,ls_change4,ls_t16fw5,ls_change5,
           ls_t16fw6,ls_change6.
  ENDLOOP.

  SORT lt_approval BY banfn.
ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_t16fs_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_t16fs_data .
  IF lt_valid IS NOT INITIAL.
    SELECT frggr,frgsx,frgc1,frgc2,
           frgc3,frgc4,frgc5,frgc6
    FROM t16fs
    FOR ALL ENTRIES IN @lt_valid
    WHERE frggr = @lt_valid-frggr AND
          frgsx = @lt_valid-frgst
    INTO TABLE @lt_t16fs.
  ENDIF .
ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_t16fw_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_t16fw_data .
  IF lt_valid IS NOT INITIAL.
    SELECT t16fw~frggr,t16fw~frgco,t16fw~objid,
           user_addr~name_textc
    FROM t16fw
    LEFT OUTER JOIN user_addr ON t16fw~objid EQ user_addr~bname "#EC CI_BUFFJOIN
    FOR ALL ENTRIES IN @lt_valid
    WHERE t16fw~frggr EQ @lt_valid-frggr
    INTO TABLE @lt_t16fw.
  ENDIF .
ENDFORM.
*&---------------------------------------------------------------------*
*& Form Change_details
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM Change_details .
  IF lt_valid IS NOT INITIAL.
    SELECT cdpos~objectid,cdpos~changenr,cdpos~value_new,
           cdhdr~udate
    FROM  cdpos
    INNER JOIN cdhdr ON cdpos~objectid = cdhdr~objectid AND
                        cdpos~changenr = cdhdr~changenr
    FOR ALL ENTRIES IN @lt_valid
    WHERE cdpos~objectid = @lt_valid-objectid
      AND cdpos~value_new LIKE 'X%'
    INTO TABLE @lt_change.
  ENDIF .

  SORT lt_change BY objectid changenr.
ENDFORM.

FORM get_data .
  "--> sort the Buffer table contail valid Document Number.
  SORT valid_belnr BY belnr.

  IF valid_belnr IS NOT INITIAL.

    "--> get the G/L Account details of Document Number .
    SELECT bseg~bukrs,bseg~belnr,bseg~gjahr,bseg~koart,bseg~shkzg,
           bseg~gsber,bseg~dmbtr,bseg~wrbtr,bseg~hkont,bseg~lifnr,
           bseg~hbkid,bseg~hktid,
           t012k~bankn,t012k~bkont,
           t012t~text1,
           t012~banks,t012~bankl,
           bnka~banka,bnka~provz,bnka~stras,bnka~ort01,bnka~brnch,
           skat~saknr,skat~txt50
    FROM bseg
    LEFT OUTER JOIN t012k ON t012k~hbkid = bseg~hbkid AND t012k~hktid = bseg~hktid
    LEFT OUTER JOIN t012t ON t012t~hbkid = bseg~hbkid AND t012t~hktid = bseg~hktid AND  t012t~spras = @sy-langu "#EC CI_BUFFJOIN
    LEFT OUTER JOIN t012  ON t012~bukrs  = bseg~bukrs AND t012~hbkid  = bseg~hbkid
    LEFT OUTER JOIN bnka  ON bnka~banks  = t012~banks AND bnka~bankl  = t012~bankl
    LEFT OUTER JOIN skat  ON skat~saknr  = bseg~hkont AND skat~spras  = @sy-langu "#EC CI_BUFFJOIN
    INTO CORRESPONDING FIELDS OF TABLE @lt_rtgs_gl
    FOR ALL ENTRIES IN @valid_belnr
    WHERE bseg~bukrs = @valid_belnr-bukrs AND
          bseg~belnr = @valid_belnr-belnr AND
          bseg~gjahr = @valid_belnr-gjahr AND
          bseg~koart = 'S'.

    "--> get the vendor details of Document Number .
    SELECT bseg~bukrs,bseg~belnr,bseg~gjahr,bseg~koart,bseg~shkzg,
           bseg~gsber,bseg~dmbtr,bseg~wrbtr,bseg~hkont,bseg~lifnr,bseg~sgtxt,
           lfa1~name1,lfa1~land1,
           lfbk~bankl,lfbk~bankn,lfbk~bkont,
           bnka~banka,bnka~provz,bnka~stras,bnka~ort01,bnka~brnch
    FROM bseg
    LEFT OUTER JOIN lfa1  ON lfa1~lifnr  = bseg~lifnr
    LEFT OUTER JOIN lfbk  ON lfbk~lifnr  = bseg~lifnr AND lfbk~banks  = lfa1~land1
    LEFT OUTER JOIN bnka  ON bnka~banks  = lfa1~land1 AND bnka~bankl  = lfbk~bankl
    INTO CORRESPONDING FIELDS OF  TABLE  @lt_rtgs_vendor
    FOR ALL ENTRIES IN @valid_belnr
    WHERE bseg~bukrs = @valid_belnr-bukrs AND
          bseg~belnr = @valid_belnr-belnr AND
          bseg~gjahr = @valid_belnr-gjahr AND
          bseg~koart = 'K'.
  ENDIF.
ENDFORM.

FORM process_data .

  SORT: lt_rtgs_gl BY belnr,
        lt_rtgs_vendor BY belnr.

  LOOP AT valid_belnr INTO DATA(ls_valid).

    ls_rtgs_final-belnr = ls_valid-belnr.
    ls_rtgs_final-gjahr = ls_valid-gjahr.
    ls_rtgs_final-bukrs = ls_valid-bukrs.
    ls_rtgs_final-budat = ls_valid-budat.
    ls_rtgs_final-rqst_person    = TEXT-010.


*-->Get the requested bank details.
    READ TABLE lt_rtgs_gl INTO DATA(ls_gl) WITH KEY belnr = ls_valid-belnr koart = 'S' shkzg = 'H'.
    IF sy-subrc EQ 0.
      ls_rtgs_final-rqst_bankn     = ls_gl-bankn.
      ls_rtgs_final-rest_ifsc_code = ls_gl-bankl.
      ls_rtgs_final-rqst_banka     = ls_gl-banka.
      ls_rtgs_final-rest_add_brnch = ls_gl-brnch.
      ls_rtgs_final-rqst_add_stras = ls_gl-stras.
      ls_rtgs_final-rqst_add_ort01 = ls_gl-ort01.

      IF ls_gl-dmbtr IS NOT INITIAL.
        ls_rtgs_final-amount         = ls_gl-dmbtr.
      ELSE.
        ls_rtgs_final-amount         = ls_gl-wrbtr.
      ENDIF.

      "--> Convert amount into WORD.
      IF ls_rtgs_final-amount IS NOT INITIAL.
        CALL FUNCTION 'HR_IN_CHG_INR_WRDS'
          EXPORTING
            amt_in_num         = CONV pc207-betrg( ls_rtgs_final-amount )
          IMPORTING
            amt_in_words       = lv_amount_word
          EXCEPTIONS
            data_type_mismatch = 1
            OTHERS             = 2.
        IF sy-subrc <> 0.
          " Implement suitable error handling here
        ENDIF.

        TRANSLATE lv_amount_word TO LOWER CASE .
        SPLIT lv_amount_word AT space INTO TABLE DATA(itab) .
        LOOP AT itab ASSIGNING  FIELD-SYMBOL(<temp>) .
          lv_word = <temp> .
          TRANSLATE lv_word+0(1) TO UPPER CASE .
          CONCATENATE lv_result lv_word INTO lv_result SEPARATED BY space .
        ENDLOOP .
        CLEAR lv_amount_word.
        lv_amount_word = lv_result.
        CLEAR: lv_word,lv_result,itab.

        ls_rtgs_final-amount_w = |{ lv_amount_word } Only| .
      ENDIF.

      "--> Get the REF: Detail need to be shown in Print.
      IF ls_gl-txt50 IS NOT INITIAL.
        IF ls_gl-txt50 CA 'cc' OR ls_gl-txt50 CA 'CC'.
          ls_rtgs_final-print_ref =  TEXT-012.
        ELSEIF ls_gl-txt50 CA 'ca' OR ls_gl-txt50 CA 'CA'.
          ls_rtgs_final-print_ref =  TEXT-013.
        ENDIF.
      ENDIF.
    ENDIF.

*--> Get Benificiary Bank details.
*---> first check wheather any supplier detail availble .
    "--> Here i have taken one Variable "LV_BENIFICIARY_VENDOR_FLAG", It indicates if any Vendor for the payment is available for this document, then it contain value as 'X'.
    CLEAR:lv_benificiary_vendor_flag.
    REFRESH : temp_rtgs_vendor.
    READ TABLE lt_rtgs_vendor INTO DATA(ls_vendor) WITH KEY belnr = ls_valid-belnr koart = 'K' shkzg = 'S'.
    IF sy-subrc EQ 0.
      lv_benificiary_vendor_flag = 'X'.

      SELECT * FROM @lt_rtgs_vendor AS vendor
      WHERE belnr = @ls_valid-belnr AND koart = 'K' AND shkzg = 'S' INTO TABLE @temp_rtgs_vendor. "#EC CI_NOORDER
      DESCRIBE TABLE temp_rtgs_vendor LINES DATA(lv_count).

      IF  lv_count EQ 1.
        IF r_rtgs EQ TEXT-003  .
          ls_rtgs_final-print_subject  = TEXT-011.
        ELSEIF r_neft EQ TEXT-003.
          ls_rtgs_final-print_subject  = TEXT-016.
        ENDIF.

        ls_rtgs_final-benif_count      = lv_count.
        ls_rtgs_final-benif_name       = ls_vendor-name1.
        ls_rtgs_final-benif_bankn      = ls_vendor-bankn.
        ls_rtgs_final-benif_ifsc_code  = ls_vendor-bankl.
        ls_rtgs_final-benif_banka      = ls_vendor-banka.
        ls_rtgs_final-benif_add_stras  = ls_vendor-stras.
        ls_rtgs_final-benif_add_ort01  = ls_vendor-ort01.
        ls_rtgs_final-benif_add_brnch  = ls_vendor-brnch.
        IF ls_vendor-bkont  = '10'  .
          ls_rtgs_final-benif_acc_type = 'Savings account'.
        ELSEIF ls_vendor-bkont  = '11'.
          ls_rtgs_final-benif_acc_type = 'Current account'.
        ENDIF.
        "ls_rtgs_final-benif_acc_type
        ls_rtgs_final-print_purpose    = TEXT-015.
        ls_rtgs_final-peint_footer_msg = 'Please remit the amount as per the aforesaid details, by debiting our account for the amount of remittance plus your charges.'.

      ELSEIF lv_count GT 1.
        IF r_rtgs EQ TEXT-003  .
          ls_rtgs_final-print_subject  = TEXT-014.
        ELSEIF r_neft EQ TEXT-003.
          ls_rtgs_final-print_subject  = TEXT-017.
        ENDIF.
        ls_rtgs_final-benif_count      = lv_count.

        CLEAR: lv_sl_no.
        SORT temp_rtgs_vendor BY belnr lifnr.
        LOOP AT temp_rtgs_vendor INTO DATA(wa_rtgs).

          lv_sl_no = lv_sl_no + 1.
          ls_rtgs_bulk_final-sl_no           = lV_sl_no.
          ls_rtgs_bulk_final-belnr           = wa_rtgs-belnr.
          ls_rtgs_bulk_final-koart           = wa_rtgs-koart.
          ls_rtgs_bulk_final-shkzg           = wa_rtgs-shkzg.
          ls_rtgs_bulk_final-sb_ca_ind       = wa_rtgs-bkont.
          ls_rtgs_bulk_final-benif_ifsc      = wa_rtgs-bankl.
          ls_rtgs_bulk_final-benif_acc       = wa_rtgs-bankn.
          ls_rtgs_bulk_final-benif_name      = wa_rtgs-name1.
          ls_rtgs_bulk_final-remitter_name   = wa_rtgs-banka.
          ls_rtgs_bulk_final-lifnr           = wa_rtgs-lifnr.
          ls_rtgs_bulk_final-purpose         = wa_rtgs-sgtxt.
          IF wa_rtgs-dmbtr IS NOT INITIAL.
            ls_rtgs_bulk_final-remit_amount  = wa_rtgs-dmbtr.
          ELSE.
            ls_rtgs_bulk_final-remit_amount  = wa_rtgs-wrbtr.
          ENDIF.
          IF wa_rtgs-shkzg = 'S'.
            ls_rtgs_bulk_final-dr_cr_ind = 'C'.
          ELSEIF wa_rtgs-shkzg = 'H'.
            ls_rtgs_bulk_final-dr_cr_ind = 'D'.
          ENDIF.

          APPEND ls_rtgs_bulk_final TO lt_rtgs_bulk_final.
          CLEAR: ls_rtgs_bulk_final,wa_rtgs.
        ENDLOOP.
      ENDIF.
    ENDIF.

*---> if supplier detail is not availble then check and show the supplier GL acount detail if availble.
    CLEAR:lv_count,ls_gl.
    IF lv_benificiary_vendor_flag NE 'X'.
      READ TABLE lt_rtgs_gl INTO LS_gl WITH KEY belnr = ls_valid-belnr koart = 'S' shkzg = 'S'.
      IF sy-subrc EQ 0.
        SELECT * FROM @lt_rtgs_GL AS gl
        WHERE belnr = @ls_valid-belnr AND koart = 'S' AND  shkzg = 'S' INTO TABLE @temp_rtgs_gl. "#EC CI_NOORDER
        DESCRIBE TABLE temp_rtgs_GL LINES lv_count.

        IF  lv_count EQ 1.
          IF r_rtgs EQ TEXT-003  .
            ls_rtgs_final-print_subject  = TEXT-011.
          ELSEIF r_neft EQ TEXT-003.
            ls_rtgs_final-print_subject  = TEXT-016.
          ENDIF.
          ls_rtgs_final-benif_count      = lv_count.
          ls_rtgs_final-benif_name       = ls_gl-text1.
          ls_rtgs_final-benif_bankn      = ls_gl-bankn.
          ls_rtgs_final-benif_ifsc_code  = ls_gl-bankl.
          ls_rtgs_final-benif_banka      = ls_gl-banka.
          ls_rtgs_final-benif_add_stras  = ls_gl-stras.
          ls_rtgs_final-benif_add_ort01  = ls_gl-ort01.
          ls_rtgs_final-benif_add_brnch  = ls_gl-brnch.
          ls_rtgs_final-benif_acc_type   = ls_rtgs_final-print_ref.

          ls_rtgs_final-print_purpose    = TEXT-015.
          ls_rtgs_final-peint_footer_msg = 'Please remit the amount as per the aforesaid details, by debiting our account for the amount of remittance plus your charges.'.

        ELSEIF lv_count GT 1.
          IF r_rtgs EQ TEXT-003  .
            ls_rtgs_final-print_subject  = TEXT-014.
          ELSEIF r_neft EQ TEXT-003.
            ls_rtgs_final-print_subject  = TEXT-017.
          ENDIF.
          ls_rtgs_final-benif_count      = lv_count.

          CLEAR: lv_sl_no.
          SORT temp_rtgs_gl BY belnr hkont.
          LOOP AT temp_rtgs_GL INTO wa_rtgs.
            lv_sl_no = lv_sl_no + 1.
            ls_rtgs_bulk_final-sl_no           = lV_sl_no.
            ls_rtgs_bulk_final-belnr           = wa_rtgs-belnr.
            ls_rtgs_bulk_final-koart           = wa_rtgs-koart.
            ls_rtgs_bulk_final-shkzg           = wa_rtgs-shkzg.
            ls_rtgs_bulk_final-sb_ca_ind       = wa_rtgs-bkont.
            ls_rtgs_bulk_final-benif_ifsc      = wa_rtgs-bankl.
            ls_rtgs_bulk_final-benif_acc       = wa_rtgs-bankn.
            ls_rtgs_bulk_final-benif_name      = wa_rtgs-text1.
            ls_rtgs_bulk_final-remitter_name   = wa_rtgs-banka.
            ls_rtgs_bulk_final-purpose         = wa_rtgs-sgtxt.
            ls_rtgs_bulk_final-hkont           = wa_rtgs-hkont.
            IF wa_rtgs-dmbtr IS NOT INITIAL.
              ls_rtgs_bulk_final-remit_amount  = wa_rtgs-dmbtr.
            ELSE.
              ls_rtgs_bulk_final-remit_amount  = wa_rtgs-wrbtr.
            ENDIF.
            IF wa_rtgs-shkzg = 'S'.
              ls_rtgs_bulk_final-dr_cr_ind = 'C'.
            ELSEIF wa_rtgs-shkzg = 'H'.
              ls_rtgs_bulk_final-dr_cr_ind = 'D'.
            ENDIF.

            APPEND ls_rtgs_bulk_final TO lt_rtgs_bulk_final.
          ENDLOOP.
        ENDIF.
      ENDIF.
    ENDIF.

    APPEND ls_rtgs_final TO lt_rtgs_final.
  ENDLOOP.

  SORT: lt_rtgs_final BY belnr,
        lt_rtgs_bulk_final BY belnr.
ENDFORM.

        ENDIF.

          EXCEPTIONS
            formatting_error = 1
            internal_error   = 2

        IF sy-subrc <> 0.
          " Implement suitable error handling here
        ENDIF.

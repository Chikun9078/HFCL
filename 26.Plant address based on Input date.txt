FUNCTION zmmfm_po_plant_address.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_WERKS) TYPE  WERKS_D
*"     REFERENCE(IV_DATE) TYPE  ERDAT
*"     REFERENCE(IV_TIME) TYPE  ERZET
*"  EXPORTING
*"     REFERENCE(ES_ADRC) TYPE  ADRC
*"----------------------------------------------------------------------
  "-->Local object declaration .
  TYPES:BEGIN OF ty_field,
          field TYPE char15,
        END OF ty_field.

  DATA: lv_objectid   TYPE cdobjectv,
        lt_field      TYPE TABLE OF ty_field WITH HEADER LINE,
        ls_adrc       TYPE adrc.

  FIELD-SYMBOLS: <fs_ADRc> LIKE ls_adrc,
                 <fs>      TYPE any.

  SELECT SINGLE werks,adrnr FROM t001w INTO @DATA(ls_t001w) WHERE werks EQ @iv_werks.
  IF sy-subrc EQ 0.
    lv_objectid = |CA01{ ls_t001w-adrnr }| .

    SELECT objectclas,objectid,changenr,udate,utime,username
    FROM cdhdr INTO TABLE @DATA(lt_cdhdr)
    WHERE objectclas = 'ADRESSE' AND objectid   = @lv_objectid AND ( udate GT @iv_date OR ( udate = @iv_date AND utime >= @iv_time ) ).

    IF sy-subrc NE 0.
      SELECT SINGLE * FROM adrc INTO ls_adrc WHERE addrnumber EQ ls_t001w-adrnr.
    ELSE.
      SORT lt_cdhdr ASCENDING BY udate utime.
      IF  lt_cdhdr IS NOT INITIAL.
        SELECT * FROM cdpos INTO TABLE @DATA(lt_cdpos)
        FOR ALL ENTRIES IN @lt_cdhdr
        WHERE objectclas = @lt_cdhdr-objectclas AND objectid = @lt_cdhdr-objectid AND changenr = @lt_cdhdr-changenr   .
        IF sy-subrc EQ 0.

          "--> Fetch the current Address Details.
          SELECT SINGLE * FROM adrc INTO ls_adrc WHERE addrnumber EQ ls_t001w-adrnr.
          IF sy-subrc EQ 0.
            ASSIGN ls_adrc TO <fs_adrc>.
          ENDIF.

          "--> Update the old address.
          SORT lt_cdpos BY  fname changenr ASCENDING.
          DELETE ADJACENT DUPLICATES FROM lt_cdpos COMPARING fname.

          LOOP AT lt_cdpos ASSIGNING FIELD-SYMBOL(<fs_field>) .
            ASSIGN COMPONENT <fs_field>-fname OF STRUCTURE <fs_adrc> TO <fs>.
            IF <fs> IS ASSIGNED.
              <fs> = <fs_field>-value_old .
              UNASSIGN <fs>.
            ENDIF.
          ENDLOOP.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

  "--> Assigning to Exporting Variable.
  IF  ls_adrc IS NOT INITIAL.
    es_adrc = ls_adrc.
  ENDIF.

ENDFUNCTION.

FORM process_data.
  IF lt_vbrk1[] IS NOT INITIAL.
    LOOP AT lt_vbrk1  ASSIGNING <ls_vbrk1>.
      IF   <ls_vbrk1> IS ASSIGNED.
        MOVE-CORRESPONDING <ls_vbrk1> TO ls_data.
        CONCATENATE <ls_vbrk1>-name1  <ls_vbrk1>-name2 INTO ls_data-name1 SEPARATED BY ' '.
        APPEND ls_data TO lt_data.
        CLEAR ls_data.
      ENDIF.
    ENDLOOP.
    SORT lt_data BY vbeln belnr.
    DELETE ADJACENT DUPLICATES FROM lt_data COMPARING vbeln belnr.
    PERFORM field_catlog.
    PERFORM display.
  ENDIF.
  IF lt_vbrk[] IS NOT INITIAL.
    LOOP AT lt_vbrk ASSIGNING <ls_vbrk>.
      IF <ls_vbrk> IS ASSIGNED.
        MOVE-CORRESPONDING <ls_vbrk> TO ls_data.
        CONCATENATE <ls_vbrk>-name1  <ls_vbrk>-name2 INTO ls_data-name1 SEPARATED BY ' '.
        APPEND ls_data TO lt_data.
        CLEAR ls_data.
      ENDIF.
    ENDLOOP.
    SORT lt_data BY vbeln belnr.
    DELETE ADJACENT DUPLICATES FROM lt_data COMPARING vbeln belnr.
    PERFORM field_catlog.
    PERFORM display.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& Form BATCH_CACULATION
*&---------------------------------------------------------------------*
FORM batch_caculation ##CALLED.
**" Here parallel cursor is used to calculate the informations from the batch items and update the same in main item column.
**" LT_FINAL is final iternal table after calculations . LT_FINAL_TEMP is sub ordinate final table.
  DATA(lt_final_temp) = lt_final[].
  LOOP AT lt_final INTO ls_final.
    IF ls_final-posnr = ls_final-uecha.                                                 " Check for if it's main line item or item having no batch split
      LOOP AT lt_final_temp INTO DATA(ls_final_temp) WHERE uecha = ls_final-posnr ##INTO_OK.      " Will iterate only the batchsplit items
        IF ls_final_temp-posnr NE ls_final_temp-uecha.                                  " Checking if iteration is batch item or main item .
          " If batch item then will do calculations
          ls_final-fkimg += ls_final_temp-fkimg.

          ls_final-basic_unit_price         += ls_final_temp-basic_unit_price.
          ls_final-freight_charges          += ls_final_temp-freight_charges.
          ls_final-packing_charges          += ls_final_temp-packing_charges.
          ls_final-insurance_charges        += ls_final_temp-insurance_charges.
          ls_final-tax_unit_rate            += ls_final_temp-tax_unit_rate.
          ls_final-total_taxable_amount     += ls_final_temp-total_taxable_amount.
          ls_final-cgst_amount              += ls_final_temp-cgst_amount.
          ls_final-total_tax_amount         += ls_final_temp-total_tax_amount.
          ls_final-total_amount_incl_tax    += ls_final_temp-total_amount_incl_tax.
          ls_final-tcs                      += ls_final_temp-tcs.
          ls_final-tax_sub_to_rev           += ls_final_temp-tax_sub_to_rev.
          ls_final-discount                 += ls_final_temp-discount.
          ls_final-igst_amount              += ls_final_temp-igst_amount.
        ENDIF.
      ENDLOOP.
    ELSE.
      CONTINUE.  " Skipping the iteration as it might a non batchsplit material or might be a main line item
    ENDIF.
    READ TABLE lt_final ASSIGNING FIELD-SYMBOL(<fs_final>) WITH KEY posnr = ls_final-posnr.
    IF sy-subrc = 0.                                                                    " Updating calculations in the final table
      <fs_final>-fkimg                    = ls_final-fkimg.
      <fs_final>-tax_unit_rate            = ls_final-tax_unit_rate.
      <fs_final>-igst_amount              = ls_final-igst_amount.
      <fs_final>-basic_unit_price         = ls_final-basic_unit_price.
      <fs_final>-freight_charges          = ls_final-freight_charges.
      <fs_final>-packing_charges          = ls_final-packing_charges.
      <fs_final>-insurance_charges        = ls_final-insurance_charges.
      <fs_final>-total_taxable_amount     = ls_final-total_taxable_amount.
      <fs_final>-cgst_amount              = ls_final-cgst_amount.
      <fs_final>-total_tax_amount         = ls_final-total_tax_amount.
      <fs_final>-total_amount_incl_tax    = ls_final-total_amount_incl_tax.
      <fs_final>-tcs                      = ls_final-tcs.
      <fs_final>-tax_sub_to_rev           = ls_final-tax_sub_to_rev.
      <fs_final>-discount                 = ls_final-discount.
      <fs_final>-del_flag = abap_true.                                                  " Marking batch items to be deleted
    ENDIF.
  ENDLOOP.
  DELETE lt_final WHERE del_flag IS INITIAL.                                            " Deleting batch items from final table
ENDFORM.

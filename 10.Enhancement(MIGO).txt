    CONSTANTS : lc_bwart    TYPE bwart VALUE '101',                          "--> Movement Type.
                lc_refdoc   TYPE refdoc VALUE 'R01',                         "--> Document Reference.
                lc_godynpro TYPE fieldname VALUE'(SAPLMIGO)GODYNPRO'.

    FIELD-SYMBOLS:<lfs_godynpro> TYPE godynpro.

    DATA: ls_mseg TYPE mseg.

    ls_mseg = CORRESPONDING #( is_mseg ).
    ASSIGN :(lc_godynpro) TO <lfs_godynpro> .

    SELECT SINGLE ebeln,ebelp,elikz
    FROM ekpo
    INTO @DATA(ls_ekpo)
    WHERE ebeln = @ls_mseg-ebeln AND
          ebelp = @ls_mseg-ebelp.
    IF sy-subrc EQ 0.
      IF ls_mseg-bwart EQ lc_bwart AND <lfs_godynpro>-refdoc EQ lc_refdoc.
        IF ls_ekpo IS NOT INITIAL.
          IF ls_ekpo-elikz NE ' '.
            MESSAGE |Delivery completed for item Number:{ ls_ekpo-ebelp }.| TYPE 'E'.
          ENDIF.
        ENDIF.
      ENDIF.

      CLEAR :ls_mseg,ls_ekpo.
    ENDIF.

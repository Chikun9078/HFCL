FORM process_data .
  LOOP AT lt_ekko INTO DATA(ls_ekko).
    LOOP AT  lt_ekpo INTO DATA(ls_ekpo).
      IF ls_ekko-ebeln EQ ls_ekpo-ebeln.
        READ TABLE lt_po_order INTO DATA(ls_po_order) WITH  KEY ebeln = ls_ekpo-ebeln
                                                          ebelp = ls_ekpo-ebelp BINARY SEARCH.
        IF sy-subrc EQ 0.
          temp_final-po_doc_type = ls_po_order-bsart.
          temp_final-po_no       = ls_po_order-ebeln.
          temp_final-llief       = |{  ls_po_order-llief ALPHA = OUT }|.
          temp_final-werks       = ls_po_order-werks.
          temp_final-po_date     = ls_po_order-aedat.
          temp_final-po_ebelp    = ls_po_order-ebelp.
          temp_final-matnr       = ls_po_order-matnr.
          temp_final-txz01       = ls_po_order-txz01.
          temp_final-po_menge    = ls_po_order-menge.
          temp_final-po_meins    = ls_po_order-meins.
          temp_final-po_netwr    = ls_po_order-netwr.
          temp_final-kbetr       = ls_po_order-kbetr.

          READ TABLE lt_sales_order INTO DATA(ls_sales_order) WITH KEY bstnk = ls_ekko-cust_ref
                                                                 posnr = ls_ekpo-ebelp.
          IF  sy-subrc EQ 0.
            temp_final-auart  = ls_sales_order-auart.
            temp_final-vbeln  = |{ ls_sales_order-vbeln ALPHA = OUT }|.
            temp_final-posnr  = ls_sales_order-posnr.
            temp_final-netwr  = ls_sales_order-netwr.
            temp_final-kwmeng = ls_sales_order-kwmeng.
            temp_final-bstnk  = ls_sales_order-bstnk.
          ENDIF.

          LOOP AT lt_inv_grn INTO DATA(ls_inv_grn).
            IF ls_inv_grn-ebeln = ls_po_order-ebeln AND ls_inv_grn-ebelp = ls_po_order-ebelp AND ls_inv_grn-matnr = ls_po_order-matnr.
              ls_stock_result = CORRESPONDING #( temp_final ).
              ls_stock_result-inv_num     = |{  ls_inv_grn-inv_num ALPHA = OUT }|.
              ls_stock_result-inv_date    = ls_inv_grn-inv_date.
              ls_stock_result-inv_qty     = ls_inv_grn-inv_qty.
              ls_stock_result-inv_value   = ls_inv_grn-inv_value.
              ls_stock_result-grn_num     = ls_inv_grn-grn_num.
              ls_stock_result-grn_date    = ls_inv_grn-grn_date.
              ls_stock_result-grn_qty     = ls_inv_grn-grn_qty.
              ls_stock_result-received_val = ls_inv_grn-received_val.
              ls_stock_result-open_qty    = ls_inv_grn-open_qty.

              APPEND ls_stock_result TO lt_stock_result.
            ENDIF.
            CLEAR: ls_stock_result.
          ENDLOOP.
        ENDIF.
        CLEAR:ls_po_order,ls_sales_order,ls_inv_grn,temp_final.
      ENDIF.
      CLEAR ls_ekpo.
    ENDLOOP.
    CLEAR: ls_ekko.
  ENDLOOP.
ENDFORM.

FORM purchase_order_data .
  PERFORM get_markrt_price_data.

  IF  lt_ekko IS NOT INITIAL.
    LOOP AT lt_ekko INTO DATA(ls_ekko).
      ls_ekko-cust_ref = LS_ekko-ebeln.
      MODIFY lt_ekko FROM ls_ekko.
    ENDLOOP.
  ENDIF.

  CLEAR: ls_ekko.
  LOOP AT lt_ekko INTO ls_ekko.
    LOOP AT lt_ekpo INTO DATA(ls_ekpo).
      IF  ls_ekpo-ebeln = ls_ekko-ebeln.

        ls_po-bsart = ls_ekko-bsart.
        ls_po-ebeln = ls_ekko-ebeln.
        ls_po-aedat = ls_ekko-aedat.
        ls_po-llief = ls_ekko-llief.
        ls_po-werks = ls_ekpo-werks.
        ls_po-ebelp = ls_ekpo-ebelp.

        ls_po-menge = ls_ekpo-menge.
        ls_po-meins = ls_ekpo-meins.
        ls_po-netwr = ls_ekpo-netwr.
        READ TABLE lt_market_price INTO DATA(ls_market) WITH KEY knumv = ls_ekko-knumv

          ls_po-kbetr = ls_market-kbetr.
        ENDIF.
        APPEND ls_po TO lt_po_order.
      ENDIF.
      CLEAR: ls_ekpo,ls_market.
    ENDLOOP.
    CLEAR: ls_ekko.

FORM sales_order_data .
  PERFORM get_vbak_data.               "--> Get Sales order Header Data.
  PERFORM get_vbap_data.               "--> Get Sales Order Item Data.

  LOOP AT lt_ekko INTO DATA(ls_ekko).
    LOOP AT lt_vbak INTO DATA(ls_vbak).
      IF ls_vbak-bstnk = ls_ekko-cust_ref.

            ls_sales-auart  = ls_vbak-auart.
            ls_sales-bstnk  = ls_ekko-cust_ref.
            ls_sales-vbeln  = ls_vbap-vbeln.
            ls_sales-posnr  = ls_vbap-posnr.
            ls_sales-netwr  = ls_vbap-netwr.
            ls_sales-kwmeng = ls_vbap-kwmeng.

            APPEND ls_sales TO lt_sales_order.
          ENDIF.
        ENDLOOP.
        CLEAR:ls_vbap.
      ENDIF.
      CLEAR:ls_vbak.
    ENDLOOP.
    CLEAR: ls_ekko.
  ENDLOOP.

  SORT lt_sales_order BY bstnk vbeln posnr.
ENDFORM.

FORM invoice_grn_data .
*---------------------------------------------------------------
*--> Local Declaration for the subroutine
  DATA: lv_open_qty TYPE ekpo-menge.

*---------------------------------------------------------------
  PERFORM invoice_data.
  PERFORM grn_data.

  LOOP AT lt_ekko INTO DATA(ls_ekko).
    LOOP AT lt_ekpo INTO DATA(ls_ekpo).
      lv_open_qty = ls_ekpo-menge.
      IF ls_ekpo-ebeln EQ ls_ekko-ebeln.
        LOOP AT lt_invoice INTO DATA(ls_invoice).
          IF ls_invoice-ebeln = ls_ekpo-ebeln AND ls_invoice-ebelp = ls_ekpo-ebelp AND ls_invoice-matnr = ls_ekpo-matnr.
*            lv_open_qty = lv_open_qty - ls_invoice-fkimg.
            ls_inv_grn-ebeln      = ls_invoice-ebeln.
            ls_inv_grn-ebelp      = ls_invoice-ebelp.
            ls_inv_grn-matnr      = ls_invoice-matnr.
            ls_inv_grn-inv_num    = ls_invoice-vbeln.
            ls_inv_grn-inv_date   = ls_invoice-fkdat.
            ls_inv_grn-inv_qty    = ls_invoice-fkimg.
            ls_inv_grn-inv_value  = ls_invoice-netwr.
*            ls_inv_grn-open_qty     = lv_open_qty .

            READ TABLE lt_grn INTO DATA(ls_grn) WITH  KEY ebeln = ls_invoice-ebeln
                                                          ebelp = ls_invoice-ebelp
                                                          belnr = ls_invoice-lfbnr .
            IF sy-subrc EQ 0.
              lv_open_qty = lv_open_qty - ls_grn-menge.
              ls_inv_grn-grn_num      = ls_grn-belnr.
              ls_inv_grn-grn_date     = ls_grn-budat.
              ls_inv_grn-grn_qty      = ls_grn-menge.
              ls_inv_grn-received_val  = ls_grn-dmbtr.
              ls_inv_grn-open_qty     = lv_open_qty.
            ENDIF.

            APPEND ls_inv_grn TO lt_inv_grn.
          ENDIF.
          CLEAR: ls_invoice,ls_grn.
        ENDLOOP.
      ENDIF.
      CLEAR: ls_ekpo,lv_open_qty.
    ENDLOOP.
  ENDLOOP.

  SORT lt_inv_grn BY ebeln ebelp inv_num.
ENDFORM.

FORM invoice_data .
*----> local declaration for making the RBKP Database
*      table's XBLNR field type Compatable
*      with VBRK Table's Vbeln field.
*------------------------------------------------------
  TYPES: BEGIN OF ty_rbkp,
           belnr     LIKE rbkp-belnr,
           xblnr     LIKE rbkp-xblnr,
           vbeln_out LIKE vbrk-vbeln,
           vbeln_in  LIKE vbrk-vbeln,
         END OF ty_rbkp.

  DATA: lt_rbkp TYPE TABLE OF ty_rbkp.
*------------------------------------------------------

  IF lt_ekko IS NOT INITIAL.
    SELECT ebeln,ebelp,vgabe,belnr,lfbnr
    FROM ekbe
    FOR ALL ENTRIES IN @lt_ekko
    WHERE ebeln = @lt_ekko-ebeln AND
          vgabe = '2'                             " Here VGABE Fiels value 2 Indicates "Invoice Recipt"
    INTO TABLE @lt_ekbe_inv.

    SORT lt_ekbe_inv BY ebeln ebelp belnr lfbnr.
  ENDIF.

  IF lt_ekbe_inv IS NOT INITIAL.
    SELECT belnr,xblnr
    FROM rbkp
    FOR ALL ENTRIES IN @lt_ekbe_inv
    WHERE belnr = @lt_ekbe_inv-belnr
    INTO TABLE @lt_rbkp.

    LOOP AT lt_rbkp INTO DATA(wa_rbkp).
      wa_rbkp-vbeln_out = |{ wa_rbkp-xblnr ALPHA = OUT }|.
      wa_rbkp-vbeln_in  = |{  wa_rbkp-vbeln_out ALPHA = IN }|.
      MODIFY lt_rbkp FROM wa_rbkp.
    ENDLOOP.
    SORT lt_rbkp BY belnr.
  ENDIF.

  IF  lt_rbkp IS NOT INITIAL.
    SELECT vbrk~vbeln,vbrk~fkdat,vbrk~xblnr,
           vbrp~posnr,vbrp~fkimg,vbrp~netwr,vbrp~matnr
    FROM vbrk
    LEFT OUTER JOIN vbrp ON vbrk~vbeln = vbrp~vbeln
    FOR ALL ENTRIES IN @lt_rbkp
    WHERE ( vbrk~vbeln = @lt_rbkp-vbeln_in and vbrk~fksto IS INITIAL ) or
          ( vbrk~xblnr = @lt_rbkp-xblnr    AND vbrk~fksto IS INITIAL )
    INTO TABLE @DATA(lt_vbrk).
    SORT lt_vbrk BY vbeln.

    IF lt_vbrk IS NOT INITIAL.
      SELECT vbeln,fkdat,xblnr,
      SUM( fkimg ) AS fkimg,SUM( netwr ) AS netwr,matnr
      FROM @lt_vbrk AS invoice                                                                 ##ITAB_DB_SELECT
      GROUP BY vbeln,fkdat,xblnr,matnr
      INTO TABLE @DATA(lt_total_inv).
    ENDIF.
    SORT lt_total_inv BY vbeln xblnr matnr.
  ENDIF.

  LOOP AT lt_ekko INTO DATA(ls_ekko).
    LOOP AT lt_ekpo INTO DATA(ls_ekpo).
      IF ls_ekpo-ebeln EQ ls_ekko-ebeln.
        LOOP AT lt_ekbe_inv INTO DATA(ls_ekbe).
          IF ls_ekbe-ebeln EQ ls_ekpo-ebeln AND ls_ekbe-ebelp EQ ls_ekpo-ebelp.
            READ TABLE lt_rbkp INTO DATA(ls_rbkp) WITH KEY belnr = ls_ekbe-belnr BINARY SEARCH.
            IF sy-subrc EQ 0.
              READ TABLE lt_total_inv INTO DATA(ls_invc) WITH  KEY vbeln = ls_rbkp-vbeln_in
                                                                   matnr = ls_ekpo-matnr BINARY SEARCH.
              IF sy-subrc EQ 0.
                ls_invoice-ebeln = ls_ekko-ebeln.
                ls_invoice-ebelp = ls_ekpo-ebelp.
                ls_invoice-vgabe = ls_ekbe-vgabe.
                ls_invoice-belnr = ls_ekbe-belnr.
                ls_invoice-xblnr = ls_rbkp-xblnr.
                ls_invoice-lfbnr = ls_ekbe-lfbnr.
                ls_invoice-vbeln = ls_invc-vbeln.
                ls_invoice-fkdat = ls_invc-fkdat.
                ls_invoice-fkimg = ls_invc-fkimg.
                ls_invoice-netwr = ls_invc-netwr.
                ls_invoice-matnr = ls_invc-matnr.

                APPEND ls_invoice TO lt_invoice.
              ELSE.
                READ TABLE lt_total_inv INTO ls_invc WITH  KEY xblnr = ls_rbkp-xblnr
                                                               matnr = ls_ekpo-matnr BINARY SEARCH.
                IF sy-subrc EQ 0.
                  ls_invoice-ebeln = ls_ekko-ebeln.
                  ls_invoice-ebelp = ls_ekpo-ebelp.
                  ls_invoice-vgabe = ls_ekbe-vgabe.
                  ls_invoice-belnr = ls_ekbe-belnr.
                  ls_invoice-xblnr = ls_rbkp-xblnr.
                  ls_invoice-lfbnr = ls_ekbe-lfbnr.
                  ls_invoice-vbeln = ls_invc-vbeln.
                  ls_invoice-fkdat = ls_invc-fkdat.
                  ls_invoice-fkimg = ls_invc-fkimg.
                  ls_invoice-netwr = ls_invc-netwr.
                  ls_invoice-matnr = ls_invc-matnr.

                  APPEND ls_invoice TO lt_invoice.
                ENDIF.
              ENDIF.
            ENDIF.
          ENDIF.
          CLEAR:ls_ekbe,ls_rbkp,ls_invc,ls_invoice.
        ENDLOOP.
      ENDIF.
      CLEAR:ls_ekpo.
    ENDLOOP.
    CLEAR:ls_ekko.
  ENDLOOP.

  SORT lt_invoice BY ebeln xblnr fkdat .
ENDFORM.

*&---------------------------------------------------------------------*
*& Form grn_data
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM grn_data .

  IF  lt_ekbe_inv IS NOT INITIAL .
    SELECT ebeln,ebelp,vgabe,belnr,budat,menge,dmbtr
    FROM  ekbe
    FOR ALL ENTRIES IN @lt_ekbe_inv
    WHERE lfbnr = @lt_ekbe_inv-lfbnr AND
          vgabe = '1'
    INTO TABLE @DATA(lt_ekbe_grn).

    SORT lt_ekbe_grn BY ebeln ebelp belnr.
  ENDIF.

  LOOP AT lt_ekko INTO DATA(ls_ekko).
    LOOP AT lt_ekpo INTO DATA(ls_ekpo).
      IF ls_ekpo-ebeln EQ ls_ekko-ebeln.
        LOOP AT lt_ekbe_inv INTO DATA(ls_ekbe_inv).
          IF ls_ekbe_inv-ebeln EQ ls_ekpo-ebeln AND
             ls_ekbe_inv-ebelp EQ ls_ekpo-ebelp.
            READ TABLE lt_ekbe_grn INTO DATA(ls_grn) WITH KEY ebeln = ls_ekbe_inv-ebeln
                                                              ebelp = ls_ekbe_inv-ebelp
                                                              belnr = ls_ekbe_inv-lfbnr BINARY SEARCH.
            IF sy-subrc EQ 0.
              ls_grn-ebeln = ls_grn-ebeln.
              ls_grn-ebelp = ls_grn-ebelp.
              ls_grn-vgabe = ls_grn-vgabe.
              ls_grn-belnr = ls_grn-belnr.
              ls_grn-budat = ls_grn-budat.
              ls_grn-menge = ls_grn-menge.
              ls_grn-dmbtr = ls_grn-dmbtr.

              APPEND ls_grn TO lt_grn.
            ENDIF.
          ENDIF.
          CLEAR : ls_ekbe_inv,ls_grn.
        ENDLOOP.
      ENDIF.
      CLEAR: ls_ekpo.
    ENDLOOP.
    CLEAR: ls_ekko.
  ENDLOOP.
ENDFORM.

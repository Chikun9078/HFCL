                APPEND wa_essr TO it_temp.
              ENDIF.
            ELSE.
              READ TABLE it_t16fs INTO wa_t16fs WITH KEY  frgsx = wa_essr-frgsx     "Level4 pending
                                                          frgc4 = s_frgco-low.
              IF sy-subrc = 0.
                IF wa_essr-frgzu = 'XXX'.
                  APPEND wa_essr TO it_temp.
                ENDIF.
              ELSE.
                READ TABLE it_t16fs INTO wa_t16fs WITH KEY  frgsx = wa_essr-frgsx   "Level5 pending

        INTO   wa_final-eknam
        FROM   t024
        WHERE  ekgrp = wa_final-ekgrp.
      SELECT   SINGLE  name1
        INTO   wa_final-name1
        FROM   lfa1
        WHERE  lifnr = wa_final-lifnr.
      SELECT   SINGLE lwert
        INTO   wa_final-lwert
        FROM   essr
        WHERE  lblni = wa_final-lblni.
      MODIFY it_final FROM wa_final INDEX zcount TRANSPORTING lifnr name1 lwert ekgrp eknam.

FORM user_command USING p_comm LIKE sy-ucomm
                        rs_selfield TYPE slis_selfield.

  DATA ref1 TYPE REF TO cl_gui_alv_grid.
  DATA: lv_seconds TYPE sytabix.
  CLEAR:lv_seconds.

  DATA: lv_sec TYPE i ."VALUE 3685.
  DATA: lv_hh(2) TYPE n.
  DATA: lv_mm(2) TYPE n.
  DATA: lv_ss(2) TYPE n.
  DATA: lv_time(10),
        lv_val      TYPE char25,
        lv_tm       TYPE tvro-fahztd.

  BREAK deepak.
  CALL FUNCTION 'GET_GLOBALS_FROM_SLVC_FULLSCR'
    IMPORTING
      e_grid = ref1.
  CALL METHOD ref1->check_changed_data.



  CASE p_comm.

    WHEN '&DATA_SAVE'.
      LOOP AT it_final INTO wa_final WHERE chk = 'X'.

        CALL FUNCTION 'BAPI_ENTRYSHEET_RELEASE'
          EXPORTING
            entrysheet = wa_final-lblni
            rel_code   = s_frgco-low
          TABLES
            return     = it_return.
        IF it_return IS NOT INITIAL.
          LOOP AT it_return INTO wa_return WHERE type = 'E'.
            MESSAGE  wa_return-message TYPE 'E'.
          ENDLOOP.
        ELSE.
          CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
            EXPORTING
              wait = 'X'.
          WAIT UP TO 1 SECONDS.
        ENDIF.
      ENDLOOP.
      LEAVE TO SCREEN 0.

  ENDCASE.

ENDFORM.

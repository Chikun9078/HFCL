FORM process_data_delv .

  LOOP AT so_del ASSIGNING FIELD-SYMBOL(<fs_del>).
    SELECT SINGLE zstk_crt_id FROM zmm_stk_entry INTO @DATA(lv_stk_entry) WHERE zmode = 'DELIVERY' AND del_no = @<fs_del>-low  ##WARN_OK ##NEEDED.
    IF sy-subrc = 0.
      DELETE so_del INDEX sy-tabix.
    ENDIF.
  ENDLOOP.

  IF so_del[] IS INITIAL.
    MESSAGE 'Delivery no mentioned are already processed !' TYPE 'I'   ##NO_TEXT.
  ENDIF.

  IF so_del[] IS NOT INITIAL.
    SELECT lips~vbeln lips~posnr lips~matnr lips~werks lips~lgort lips~ps_psp_pnr likp~kunnr lips~lfimg lips~meins
      INTO TABLE lt_lips ##TOO_MANY_ITAB_FIELDS
       FROM likp INNER JOIN lips
      ON likp~vbeln = lips~vbeln
    WHERE lips~vbeln IN so_del AND
          lips~lgort <> ''.

    IF sy-subrc EQ 0.

      LOOP AT lt_lips ASSIGNING FIELD-SYMBOL(<fs_lips>).

        AUTHORITY-CHECK OBJECT 'ZMMA_CDEV'
         ID 'WERKS' FIELD <fs_lips>-werks
         ID 'TCODE' FIELD 'ZMM0054'.
        IF sy-subrc <> 0.
          CLEAR:<fs_lips>-werks.
        ENDIF.

      ENDLOOP.

      DELETE lt_lips WHERE werks IS INITIAL.

      IF lt_lips IS INITIAL.
        MESSAGE e166(zmm).
      ENDIF.

    ENDIF.

  ENDIF.
ENDFORM.

FORM process_data_billing .

  IF so_bill IS NOT INITIAL.
    LOOP AT so_bill ASSIGNING FIELD-SYMBOL(<fs_bill>).
      SELECT SINGLE zstk_crt_id FROM zmm_stk_entry INTO @DATA(lv_stk_entry) WHERE zmode = 'BILLING' AND bill_no = @<fs_bill>-low  ##WARN_OK ##NEEDED.
      IF sy-subrc = 0.
        DELETE so_bill INDEX sy-tabix.
      ENDIF.
    ENDLOOP.

    IF so_bill[] IS INITIAL.
      MESSAGE 'Billing no mentioned are already processed !' TYPE 'I' ##NO_TEXT.
    ENDIF.

    IF so_bill[] IS NOT INITIAL.
      SELECT vbrp~vbeln vbrp~posnr vbrp~matnr vbrp~werks vbrp~lgort vbrp~ps_psp_pnr vbrk~kunrg vbrp~fkimg vbrp~meins vbrp~pstyv
        INTO TABLE lt_vbrp    ##TOO_MANY_ITAB_FIELDS
        FROM vbrk INNER JOIN vbrp
        ON vbrk~vbeln = vbrp~vbeln
      WHERE vbrp~vbeln IN so_bill AND
            vbrp~lgort <> ''.

      IF sy-subrc EQ 0.

        LOOP AT lt_vbrp ASSIGNING FIELD-SYMBOL(<fs_vbrp>).

          AUTHORITY-CHECK OBJECT 'ZMMA_CDEV'
           ID 'WERKS' FIELD <fs_vbrp>-werks
           ID 'TCODE' FIELD 'ZMM0054'.
          IF sy-subrc <> 0.
            CLEAR:<fs_vbrp>-werks.
          ENDIF.

        ENDLOOP.

        DELETE lt_vbrp WHERE werks IS INITIAL.

        IF lt_vbrp IS INITIAL.
          MESSAGE e166(zmm).
        ENDIF.

        LOOP AT lt_vbrp INTO ls_vbrp WHERE pstyv = 'ZTPQ'.
          SELECT SINGLE vbelv, posnv, vbeln, posnn FROM vbfa INTO @DATA(ls_vbfa) ##WARN_OK "#EC CI_NOORDER "#EC CI_NOFIELD
            WHERE vbtyp_v = 'J' AND
                  vbeln = @ls_vbrp-vbeln AND
                  posnn = @ls_vbrp-posnr.

          SELECT vbeln,
                 posnr,
                 matnr,
                 werks,
                 lgort,
                 lfimg,
                 ps_psp_pnr
            FROM lips INTO TABLE @DATA(it_lips) WHERE vbeln = @ls_vbfa-vbelv AND uepos = '000010'.
          LOOP AT it_lips INTO DATA(is_lips)  ##INTO_OK.
            ls_vbrp-vbeln = is_lips-vbeln.
            ls_vbrp-posnr = is_lips-posnr.
            ls_vbrp-matnr = is_lips-matnr.
            ls_vbrp-werks = is_lips-werks.
            ls_vbrp-lgort = is_lips-lgort.
            ls_vbrp-ps_psp_pnr = is_lips-ps_psp_pnr.
            SELECT SINGLE kunnr FROM likp INTO @DATA(lv_kunnr) WHERE vbeln = @is_lips-vbeln.
            ls_vbrp-cust = lv_kunnr.
            ls_vbrp-fkimg = is_lips-lfimg.
            SELECT SINGLE meins FROM mara INTO @DATA(lv_unit) WHERE matnr = @is_lips-matnr.
            ls_vbrp-meins = lv_unit.
            ls_vbrp-pstyv = ''.
            APPEND ls_vbrp TO lt_vbrp.
          ENDLOOP.

          DELETE lt_vbrp WHERE pstyv = 'ZTPQ'.

        ENDLOOP.

      ENDIF.
    ENDIF.

  ENDIF.
ENDFORM.

FORM validate_file USING lv_valid  ##PERF_NO_TYPE.
*  DATA lv_valid_data TYPE c VALUE 'X'.
  SORT gt_file BY material.
  DATA : lv_index TYPE i.
  LOOP AT gt_file INTO gs_file.
    lv_index = sy-tabix.
    CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
      EXPORTING
        input  = gs_file-material
      IMPORTING
        output = gs_file-material.
    SELECT SINGLE matnr, meins FROM mara INTO @DATA(lv_matnr_meins) WHERE matnr = @gs_file-material.
    IF sy-subrc <> 0.
      lv_valid = ''.
      CONCATENATE 'In-correct material no' gs_file-material 'in file' INTO DATA(lv_matmsg) SEPARATED BY ''  ##NO_TEXT.
      MESSAGE lv_matmsg TYPE 'E'.

    ELSE.
      gs_file-uom = lv_matnr_meins-meins.
      MODIFY gt_file FROM gs_file TRANSPORTING uom .
    ENDIF.

    SELECT SINGLE werks FROM t001w INTO @DATA(lv_plant) WHERE werks = @gs_file-plant  ##NEEDED.
    IF sy-subrc <> 0.
      lv_valid = ''.
      CONCATENATE 'In-correct plant' gs_file-plant 'in file' INTO DATA(lv_plantmsg) SEPARATED BY ''  ##NO_TEXT.
      MESSAGE lv_plantmsg TYPE 'E'.
    ENDIF.
    SELECT SINGLE lgort FROM t001l INTO @DATA(lv_sloc) WHERE werks = @gs_file-plant AND lgort = @gs_file-storage_location   ##NEEDED.
    IF sy-subrc <> 0.
      lv_valid = ''.
      CONCATENATE 'Storage location' gs_file-storage_location 'not mapped with plant' gs_file-plant 'in file' INTO DATA(lv_slocmsg) SEPARATED BY ''  ##NO_TEXT.
      MESSAGE lv_slocmsg TYPE 'E'.
    ENDIF.

    "--------------- validate WBS------------------
    DATA: lv_wbs_ele_conv TYPE char24  ##NEEDED.
    CALL FUNCTION 'CONVERSION_EXIT_ABPSP_INPUT'
      EXPORTING
        input  = gs_file-wbs
      IMPORTING
        output = lv_wbs_ele_conv.
    SELECT SINGLE pspnr FROM prps INTO @DATA(lv_pspnr) WHERE pspnr = @lv_wbs_ele_conv  ##NEEDED.
    IF sy-subrc <> 0.
      lv_valid = ''.
      CONCATENATE 'In-correct wbs' gs_file-wbs 'in file' INTO DATA(lv_wbsmsg) SEPARATED BY '' ##NO_TEXT.
      MESSAGE lv_wbsmsg TYPE 'E'.
    ENDIF.

    CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
      EXPORTING
        input  = gs_file-customer
      IMPORTING
        output = gs_file-customer.
    SELECT SINGLE kunnr, name1 FROM kna1 INTO @DATA(lv_kunnr) WHERE kunnr = @gs_file-customer+8(10).
    IF sy-subrc <> 0.
      lv_valid = ''.
      CONCATENATE 'In-correct customer id' gs_file-customer+8(10) 'in file' INTO DATA(lv_custmsg) SEPARATED BY ''   ##NO_TEXT.
      MESSAGE lv_custmsg TYPE 'E'.
    ELSE.
      gs_file-customer_name = lv_kunnr-name1.
      MODIFY gt_file FROM gs_file INDEX lv_index TRANSPORTING customer_name.
    ENDIF.

  ENDLOOP.
ENDFORM.

FORM data_changed USING ir_data_changed TYPE REF TO cl_alv_changed_data_protocol   ##called.
  DATA ls_modi TYPE lvc_s_modi.
  DATA : lv_flag TYPE string    .
  LOOP AT ir_data_changed->mt_mod_cells INTO ls_modi  ##into_ok.

    DATA(lv_index) = ls_modi-row_id.
    READ TABLE gt_file INTO gs_file INDEX ls_modi-row_id.
    IF lv_index = ls_modi-row_id AND lv_flag = abap_true.
      CLEAR : lv_flag.
    ENDIF.

    CASE ls_modi-fieldname.
      WHEN 'REMARKS'.
        READ TABLE gt_file INTO gs_file INDEX ls_modi-row_id.
        IF sy-subrc = 0.
          ls_stk_entry-remarks = ls_modi-value.
          MODIFY gt_file FROM gs_file INDEX ls_modi-row_id TRANSPORTING remarks.
        ENDIF.
    ENDCASE.

  ENDLOOP.
  PERFORM refresh_grid.
ENDFORM.                    "data_changed
